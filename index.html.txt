function computeHealthScore(record) {
  const ms = record.milestones || [];
  const msScore = ms.length ? (ms.filter((m) => m.completedDate).length / ms.length) * 100 : 100;
  const actions = record.actions || [];
  const actionScore = actions.length ? (actions.filter((a) => a.status === "Closed").length / actions.length) * 100 : 100;
  const overduePenalty = Math.min(30, actions.filter((a) => isOverdue(a.dueDate, a.status)).length * 8);
  const evidence = record.evidence || [];
  const evidenceScore = evidence.length ? (evidence.filter((e) => e.status === "Reviewed" || e.status === "Received").length / evidence.length) * 100 : 100;
  const raw = msScore * 0.4 + actionScore * 0.35 + evidenceScore * 0.25 - overduePenalty;
  return Math.max(0, Math.min(100, Math.round(raw)));
}

const PERIODS = ["All time", "This month", "Last month", "This quarter", "Last quarter", "This financial year"];
function periodRange(period, offset = 0) {
  const now = new Date();
  let y = now.getFullYear(), qStart, monthsBack = 0;
  if (period === "All time") return null;
  if (period === "This month" || period === "Last month") {
    monthsBack = period === "Last month" ? 1 : 0;
    const start = new Date(y, now.getMonth() - monthsBack - offset, 1);
    const end = new Date(y, now.getMonth() - monthsBack - offset + 1, 0);
    return [start, end];
  }
  if (period === "This financial year") {
    // UK corporate financial year, 1 April – 31 March
    const fyStartYear = now.getMonth() >= 3 ? y : y - 1;
    const start = new Date(fyStartYear - offset, 3, 1);
    const end = new Date(fyStartYear - offset + 1, 2, 31);
    return [start, end];
  }
  const q = Math.floor(now.getMonth() / 3);
  const qBack = period === "Last quarter" ? 1 : 0;
  const start = new Date(y, (q - qBack - offset) * 3, 1);
  const end = new Date(y, (q - qBack - offset) * 3 + 3, 0);
  return [start, end];
}
function inRange(dateStr, range) {
  if (!range) return true;
  if (!dateStr) return false;
  const d = new Date(dateStr);
  return d >= range[0] && d <= range[1];
}

/* Ring chart — pure SVG, no charting library, MAG-branded, animated draw-in */
function RingChart({ segments, size = 132, thickness = 16, centerLabel }) {
  const [mounted, setMounted] = useState(false);
  useEffect(() => { const t = setTimeout(() => setMounted(true), 30); return () => clearTimeout(t); }, []);
  const total = segments.reduce((s, x) => s + x.value, 0);
  const r = (size - thickness) / 2;
  const circ = 2 * Math.PI * r;
  let offset = 0;
  const gid = React.useId ? React.useId() : `ring_${Math.random()}`;
  return (
    <div className="flex items-center gap-5 flex-wrap">
      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} style={{ flexShrink: 0, filter: "drop-shadow(0 2px 6px rgba(4,30,66,0.12))" }}>
        <defs>
          <linearGradient id={`bg-${gid}`} x1="0%" y1="0%" x2="100%" y2="100%">
            <stop offset="0%" stopColor={T.skyTint} />
            <stop offset="100%" stopColor={T.grayTint} />
          </linearGradient>
        </defs>
        <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke={`url(#bg-${gid})`} strokeWidth={thickness} />
        {total > 0 && segments.filter((s) => s.value > 0).map((s, i) => {
          const frac = s.value / total;
          const dash = frac * circ;
          const currentOffset = offset;
          const el = (
            <circle
              key={i}
              cx={size / 2} cy={size / 2} r={r} fill="none"
              stroke={s.color} strokeWidth={thickness}
              strokeLinecap="round"
              strokeDasharray={mounted ? `${Math.max(dash - 3, 0)} ${circ - dash + 3}` : `0 ${circ}`}
              strokeDashoffset={-currentOffset}
              transform={`rotate(-90 ${size / 2} ${size / 2})`}
              style={{ transition: "stroke-dasharray 800ms cubic-bezier(.4,0,.2,1)" }}
            />
          );
          offset += dash;
          return el;
        })}
        <text x="50%" y="46%" textAnchor="middle" style={{ fontSize: size < 100 ? 20 : 26, fontWeight: 800, fill: T.navy, fontFamily: "Inter, sans-serif" }}>{total}</text>
        <text x="50%" y="62%" textAnchor="middle" style={{ fontSize: 9, fontWeight: 600, letterSpacing: 0.5, fill: T.textMute, fontFamily: "Inter, sans-serif", textTransform: "uppercase" }}>{centerLabel || "total"}</text>
      </svg>
      <div className="flex flex-col gap-1.5">
        {segments.map((s) => (
          <div key={s.label} className="flex items-center gap-2 text-xs">
            <span style={{ width: 10, height: 10, borderRadius: 3, background: s.color, display: "inline-block", flexShrink: 0, boxShadow: `0 0 0 3px ${s.color}22` }} />
            <span style={{ color: T.text }}>{s.label}</span>
            <span className="font-extrabold" style={{ color: T.navy }}>{s.value}</span>
          </div>
        ))}
      </div>
    </div>
  );
}

function Delta({ current, previous, label = "vs prior period", invert = false }) {
  if (previous === 0 && current === 0) return null;
  const diff = current - previous;
  const up = diff > 0;
  const flat = diff === 0;
  const bad = invert ? !up : up;
  const color = flat ? T.gray : bad ? T.red : T.green;
  const bg = flat ? T.grayTint : bad ? T.redTint : T.greenTint;
  return (
    <span className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-xs font-bold" style={{ color, background: bg }}>
      {flat ? "→" : up ? "↑" : "↓"} {Math.abs(diff)} {label}
    </span>
  );
}

function buildAssistantContext(index, records) {
  const allRecords = Object.values(records);
  const allFindings = allRecords.flatMap((r) => r.findings.map((f) => ({ ...f, eng: r.meta.title })));
  const allActions = allRecords.flatMap((r) => r.actions.map((a) => ({ ...a, eng: r.meta.title })));
  const overdue = allActions.filter((a) => isOverdue(a.dueDate, a.status));
  const lines = [];
  lines.push(`${index.length} engagement(s) total.`);
  STAGES.forEach((s) => { const n = index.filter((e) => e.stage === s).length; if (n) lines.push(`- ${n} in ${s}`); });
  lines.push(`${allFindings.length} finding(s) raised across all engagements.`);
  lines.push(`${allActions.length} action(s) total, ${overdue.length} currently overdue.`);
  lines.push("");
  lines.push("Engagements:");
  index.forEach((e) => lines.push(`- "${e.title}" — ${e.airport}, ${e.area}, stage: ${e.stage}, lead: ${e.lead || "unassigned"}, target: ${e.targetDate || "not set"}`));
  if (overdue.length > 0) {
    lines.push("");
    lines.push("Overdue actions:");
    overdue.slice(0, 25).forEach((a) => lines.push(`- [${a.eng}] ${a.description} — owner: ${a.owner || "unassigned"}, was due ${a.dueDate}`));
  }
  const text = lines.join("\n");
  return text.length > 8000 ? text.slice(0, 8000) + "\n…(truncated)" : text;
}

          </div>
        ))}
        {loading && <div className="text-xs" style={{ color: T.textMute }}>Scanning…</div>}
        {error && <div className="text-xs font-semibold" style={{ color: T.red }}>{error}</div>}
      </div>
      <div className="flex items-center gap-2 p-3 flex-shrink-0" style={{ borderTop: `1px solid ${T.border}` }}>
        <TextInput value={input} onChange={(e) => setInput(e.target.value)} onKeyDown={(e) => { if (e.key === "Enter") send(); }} placeholder="Ask about a programme, action, or trend…" style={{ flex: 1 }} />
        <button onClick={() => send()} disabled={loading} style={{ color: T.ocean, opacity: loading ? 0.5 : 1 }}><Send size={18} /></button>
      </div>
    </div>
  );
});

function FeedbackWidget({ role }) {
  const [open, setOpen] = useState(false);
  const [name, setName] = useState(role && role !== "Audit team member" ? role : "");
  const [context, setContext] = useState("");
  const [message, setMessage] = useState("");
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  const submit = async () => {
    if (!message.trim()) return;
    setSaving(true);
    setError("");
    try {
      let list = [];
      try {
        const res = await window.storage.get("assurance:feedback", SHARED_STORAGE);
        list = res ? JSON.parse(res.value) : [];
      } catch (e) { /* no feedback saved yet */ }
      const entry = { id: uid("fb"), name: name.trim() || "Anonymous", context: context.trim(), message: message.trim(), ts: new Date().toISOString() };
      await window.storage.set("assurance:feedback", JSON.stringify([entry, ...list]), SHARED_STORAGE);
      setMessage("");
      setContext("");
      setOpen(false);
      if (_notifyToast) _notifyToast("Feedback sent — thank you");
    } catch (e) {
      setError("Couldn't save that — please try again.");
    } finally {
      setSaving(false);
    }
  };

  return (
    <>
      <button onClick={() => setOpen(true)} title="Send feedback" className="fixed flex items-center justify-center rounded-full no-print" style={{ bottom: 24, left: 24, width: 52, height: 52, background: T.teal, color: "white", zIndex: 60, boxShadow: "0 6px 20px rgba(4,30,66,0.3)" }}>
        <MessageSquarePlus size={22} />
      </button>
      {open && (
        <Modal title="Send feedback" onClose={() => setOpen(false)}>
          <div className="flex flex-col gap-3">
            <p className="text-xs" style={{ color: T.textMute }}>Found something confusing, broken, or just have a thought? This goes straight to Steve.</p>
            <Field label="Your name"><TextInput value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Conor O'Brien" /></Field>
            <Field label="What page or feature is this about? (optional)"><TextInput value={context} onChange={(e) => setContext(e.target.value)} placeholder="e.g. Action tracker" /></Field>
            <Field label="Your feedback"><TextArea rows={4} value={message} onChange={(e) => setMessage(e.target.value)} placeholder="What worked, what didn't, what confused you, what you'd want changed…" /></Field>
            {error && <p className="text-xs font-semibold" style={{ color: T.red }}>{error}</p>}
            <div className="flex justify-end gap-2"><Button variant="ghost" onClick={() => setOpen(false)}>Cancel</Button><Button onClick={submit} style={{ opacity: saving || !message.trim() ? 0.6 : 1 }}>{saving ? "Sending…" : "Send feedback"}</Button></div>
          </div>
        </Modal>
      )}
    </>
  );
}

export default function App() {
  const [loading, setLoading] = useState(true);
  const radarRef = React.useRef(null);
  const [toast, setToast] = useState(null);
  useEffect(() => {
    registerToastNotifier((msg) => setToast(msg));
  }, []);
  useEffect(() => {
    if (!toast) return;
    const t = setTimeout(() => setToast(null), 1700);
    return () => clearTimeout(t);
  }, [toast]);
  const [index, setIndex] = useState([]);
  const [records, setRecords] = useState({});
  const [settings, setSettings] = useState({ themes: DEFAULT_THEMES, orgName: "Manchester Airports Group", controlLibraries: [], adminPin: "", sharedLink: "" });
  const [role, setRole] = useState("Compliance & Assurance Lead");
  const [view, setView] = useState("home"); // home | dashboard | actions | detail | trends
  const [activeId, setActiveId] = useState(null);
  const [showNewEngagement, setShowNewEngagement] = useState(false);
  const [showImport, setShowImport] = useState(false);
  const [showSettings, setShowSettings] = useState(false);
  const [showSettingsThemes, setShowSettingsThemes] = useState(false);
  const [error, setError] = useState(null);

  const saveWithRetry = useCallback(async (key, value, attempts = 3) => {
    for (let i = 0; i < attempts; i++) {
      try {
        await window.storage.set(key, value, SHARED_STORAGE);
        return true;
      } catch (e) {
        console.error(`Storage save failed for key "${key}" (attempt ${i + 1}/${attempts}, value size ${value.length} chars):`, e);
        if (i < attempts - 1) await new Promise((r) => setTimeout(r, 400 * (i + 1)));
      }
    }
    return false;
  }, []);
  const saveIndex = useCallback(async (nextIndex) => {
    const ok = await saveWithRetry("assurance:index", JSON.stringify(nextIndex));
    if (!ok) setError({ message: "Could not save the engagement list — your last change may not persist.", retry: () => saveIndex(nextIndex) });
    else setError((e) => (e && e.message?.includes("engagement list") ? null : e));
  }, [saveWithRetry]);
  const saveRecord = useCallback(async (rec) => {
    const ok = await saveWithRetry(`assurance:engagement:${rec.meta.id}`, JSON.stringify(rec));
    if (!ok) setError({ message: `Could not save "${rec.meta.title}" — your last change may not persist.`, retry: () => saveRecord(rec) });
    else setError((e) => (e && e.message?.includes(rec.meta.title) ? null : e));
  }, [saveWithRetry]);
  const saveSettings = useCallback(async (next) => {
    const ok = await saveWithRetry("assurance:settings", JSON.stringify(next));
    if (!ok) setError({ message: "Could not save settings — your last change may not persist.", retry: () => saveSettings(next) });
  }, [saveWithRetry]);

      try {
        const idxRes = await window.storage.get("assurance:index", SHARED_STORAGE);
        setIndex(idxRes ? JSON.parse(idxRes.value) : []);
      } catch (e) { setIndex([]); }
      try {
        const sRes = await window.storage.get("assurance:settings", SHARED_STORAGE);

      } catch (e) { setSettings({ themes: DEFAULT_THEMES, orgName: "Manchester Airports Group", controlLibraries: [], adminPin: "", sharedLink: "" }); }
      setLoading(false);
    })();
  }, []);

  const loadRecord = useCallback(async (id) => {
    let rec = null;
    try {
      const res = await window.storage.get(`assurance:engagement:${id}`, SHARED_STORAGE);
      rec = res ? JSON.parse(res.value) : null;
      if (rec) setRecords((prev) => ({ ...prev, [id]: rec }));
    } catch (e) { /* not found */ }
    return rec;
  }, []);

  const ensureAllLoaded = useCallback(async () => {
    const missing = index.filter((e) => !records[e.id]);
    await Promise.all(missing.map((e) => loadRecord(e.id)));
  }, [index, records, loadRecord]);

  const openEngagement = async (id) => { setActiveId(id); setView("detail"); await loadRecord(id); };

  const updateQueueRef = React.useRef(Promise.resolve());
  const updateRecord = (id, updater) => {
    // Every update is chained after the previous one fully finishes — including
    // its storage writes — so two rapid changes (e.g. closing an action, then
    // immediately changing stage) can never land out of order and clobber
    // each other's result.
    const run = async () => {
      let nextRecord = null;
      let nextIdx = null;
      setRecords((prev) => {
        const current = prev[id];
        if (!current) {
          console.warn(`updateRecord called for engagement ${id}, but it isn't loaded in state — this update was dropped. This shouldn't happen from normal navigation; if you see this, please report the steps that led here.`);
          return prev;
        }
        nextRecord = updater(current);
        return { ...prev, [id]: nextRecord };
      });
      if (!nextRecord) return;
      setIndex((prevIdx) => {
        nextIdx = prevIdx.map((e) => e.id === id ? { ...e, title: nextRecord.meta.title, airport: nextRecord.meta.airport, area: nextRecord.meta.area, type: nextRecord.meta.type, stage: nextRecord.meta.stage, lead: nextRecord.meta.lead, targetDate: nextRecord.meta.targetDate, startDate: nextRecord.meta.startDate, priority: nextRecord.meta.priority || "Medium", updatedAt: new Date().toISOString() } : e);
        return nextIdx;
      });
      // Sequenced, not fired together — storage writes are rate-limited, and
      // two near-simultaneous calls were a real cause of save failures.
      await saveRecord(nextRecord);
      if (nextIdx) await saveIndex(nextIdx);
      setToast("Saved");
    };
    updateQueueRef.current = updateQueueRef.current.then(run, run);
    return updateQueueRef.current;
  };

  const createEngagement = async (rawMeta) => {
    const { _draftControls, _draftEvidence, _aiDrafted, ...meta } = rawMeta;
    const id = uid("eng");

      const realRef = nextControlRef(workbook, controlPrefix);

    });
    const evidence = (_draftEvidence || []).map((e) => ({ id: uid("ev"), controlRef: (e.controlDraftId && draftIdToRealRef[e.controlDraftId]) || "", description: e.description, requestedFrom: "", dateRequested: todayISO(), dueDate: "", link: "", status: "Not Requested" }));
    const selfAssessment = workbook.map((c) => ({ id: uid("sa"), ref: c.ref, control: c.description, selfRating: "In Progress", comments: "", completedBy: "" }));
    const record = { meta: { id, stage: "Planning", aiDrafted: !!_aiDrafted, ...meta, controlPrefix }, contacts: [], workbook, evidence, findings: [], actions: [], milestones: defaultMilestones(), selfAssessment, log: [] };
    const summary = { id, title: meta.title, airport: meta.airport, area: meta.area, type: meta.type, stage: "Planning", lead: meta.lead, targetDate: meta.targetDate, startDate: meta.startDate, priority: meta.priority || "Medium" };
    const nextIdx = [summary, ...index];
    setIndex(nextIdx);
    setRecords((prev) => ({ ...prev, [id]: record }));
    await saveIndex(nextIdx);
    await saveRecord(record);
    setShowNewEngagement(false);
    setActiveId(id);
    setToast("Saved");
    setView("detail");
  };

  const updateThemes = (next) => { setSettings((s) => { const merged = { ...s, themes: next }; saveSettings(merged); return merged; }); };
  const updateControlLibraries = (next) => { setSettings((s) => { const merged = { ...s, controlLibraries: next }; saveSettings(merged); return merged; }); };
  const updateOrgName = (next) => { setSettings((s) => { const merged = { ...s, orgName: next }; saveSettings(merged); return merged; }); };
  const updateAdminPin = (next) => { setSettings((s) => { const merged = { ...s, adminPin: next }; saveSettings(merged); return merged; }); };
  const updateSharedLink = (next) => { setSettings((s) => { const merged = { ...s, sharedLink: next }; saveSettings(merged); return merged; }); };

  const resetAllData = async () => {
    try {
      const listRes = await window.storage.list("assurance:", SHARED_STORAGE);
      const keys = listRes?.keys || [];
      await Promise.all(keys.map((k) => window.storage.delete(k, SHARED_STORAGE).catch(() => {})));
    } catch (e) { /* best effort */ }
    setIndex([]);
    setRecords({});
    setSettings({ themes: DEFAULT_THEMES, orgName: "Manchester Airports Group", controlLibraries: [], adminPin: "", sharedLink: "" });
    setShowSettings(false);
    setView("home");
  };

  const importData = async ({ targetEngagement, newTitle, findings }) => {
    let id = targetEngagement;
    if (targetEngagement === "__new__") {
      id = uid("eng");
      const meta = { id, title: newTitle || "Imported tracker data", airport: "MAN", area: AREAS[0], type: "Assurance Review", lead: LEADS[0], stage: "Fieldwork", startDate: todayISO(), targetDate: "", objective: "Imported from an existing Excel tracker." };
      const record = { meta, contacts: [], workbook: [], evidence: [], findings: [], actions: [], milestones: defaultMilestones(), selfAssessment: [], log: [] };
      const summary = { id, title: meta.title, airport: meta.airport, area: meta.area, type: meta.type, stage: meta.stage, lead: meta.lead, targetDate: meta.targetDate, startDate: meta.startDate, priority: meta.priority || "Medium" };
      let nextIdx = null;
      setIndex((prev) => { nextIdx = [summary, ...prev]; return nextIdx; });

      await saveIndex(nextIdx);
      await saveRecord(record);

      await loadRecord(id);

        newFindings.push({ id: f.id, ref: f.ref, title: f.title, description: f.description, riskRating: f.riskRating, theme: f.theme, rootCause: "", dateRaised: f.dateRaised });
        if (f._actionDescription) newActions.push({ id: uid("act"), findingId: f.id, description: f._actionDescription, owner: f._actionOwner, dueDate: f._actionDue, status: f._actionStatus });

    });
    setShowImport(false);
    setActiveId(id);
    setToast("Saved");
    setView("detail");
  };

  const deleteEngagement = async (id) => {
    const nextIdx = index.filter((e) => e.id !== id);
    setIndex(nextIdx);
    setRecords((prev) => { const next = { ...prev }; delete next[id]; return next; });
    try { await window.storage.delete(`assurance:engagement:${id}`, SHARED_STORAGE); } catch (e) { /* best effort */ }
    await saveIndex(nextIdx);

    setView("dashboard");

  };

    const contacts = source.contacts.map((c) => ({ ...c, id: uid("ct") }));
    const selfAssessment = workbook.map((c) => ({ id: uid("sa"), ref: c.ref, control: c.description, selfRating: "In Progress", comments: "", completedBy: "" }));

"Actions" : "Closed", startDate: g.startDate || todayISO(), targetDate: "", objective: "Imported from an existing Excel tracker." };
      const findings = g.findings.map((f) => ({ id: f.id, ref: f.ref, title: f.title, description: f.description, riskRating: f.riskRating, theme: f.theme, rootCause: "", dateRaised: f.dateRaised }));
      const record = { meta, contacts: [], workbook: [], evidence: [], findings, actions, milestones: defaultMilestones(), selfAssessment: [], log: [] };
      const summary = { id, title: meta.title, airport: meta.airport, area: meta.area, type: meta.type, stage: meta.stage, lead: meta.lead, targetDate: meta.targetDate, startDate: meta.startDate, priority: meta.priority || "Medium" };
      setRecords((prev) => ({ ...prev, [id]: record }));
      await saveRecord(record);
      newSummaries.push(summary);
      lastId = id;
    }
    // Index is saved once at the end, not once per engagement — writing the
    // full (growing) index 71 times in a tight loop is exactly the pattern
    // that trips storage rate limits.
    let nextIdx = null;
    setIndex((prev) => { nextIdx = [...newSummaries, ...prev]; return nextIdx; });
    await saveIndex(nextIdx);
    setShowImport(false);
    if (lastId) setToast(`Imported ${groups.length} engagement${groups.length !== 1 ? "s" : ""}`);
    if (lastId) { setActiveId(lastId); setView("detail"); } else { setView("dashboard"); }
  };

  if (loading) {
    return <div className="min-h-screen flex items-center justify-center" style={{ background: T.bg }}><p style={{ color: T.textMute, fontFamily: "Inter, sans-serif" }}>Loading assurance workspace…</p></div>;
  }

  return (
    <div className="min-h-screen w-full flex" style={{ background: T.bg, fontFamily: "Inter, sans-serif" }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700;800&family=JetBrains+Mono:wght@500;600&display=swap');
        @media print {
          .no-print { display: none !important; }
          .print-page { position: absolute; inset: 0; margin: 0; padding: 24px !important; }
        }
        @keyframes radar-ping-anim { 0% { transform: scale(1); opacity: 0.7; } 100% { transform: scale(1.6); opacity: 0; } }
        .radar-ping { animation: radar-ping-anim 1.8s cubic-bezier(0,0,0.2,1) infinite; }
      `}</style>

      <aside className="flex-shrink-0 flex flex-col no-print" style={{ width: 232, background: T.navy }}>
        <div className="px-5 py-6">
          <div className="flex items-center gap-2">
            <MagnifierMark size={26} ring="white" accent={T.teal} />
            <span className="text-white" style={{ fontFamily: "Arial, Helvetica, sans-serif", fontWeight: 900, fontSize: 19, letterSpacing: -0.5 }}>ASSURELY</span>
          </div>
          <div className="text-xs mt-1.5 font-bold" style={{ color: T.teal, letterSpacing: 1 }}>THE ASSURANCE INTELLIGENCE PLATFORM</div>
        </div>
        <nav className="flex flex-col gap-1 px-3 mt-2">
          <NavBtn active={view === "home"} icon={Home} label="Home" onClick={() => setView("home")} />
          <NavBtn active={view === "dashboard" || view === "detail"} icon={Compass} label="Programme overview" onClick={() => setView("dashboard")} />
          <NavBtn active={view === "actions"} icon={ListTodo} label="Action tracker" onClick={async () => { await ensureAllLoaded(); setView("actions"); }} />
          <NavBtn active={view === "trends"} icon={TrendingUp} label="Trends & themes" onClick={async () => { await ensureAllLoaded(); setView("trends"); }} />
        </nav>
        <div className="mt-auto px-4 py-4">
          <button onClick={() => setShowSettings(true)} className="w-full flex items-center gap-2 px-3 py-2.5 rounded-md text-sm font-medium" style={{ color: "#C9D6E8" }}>
            <SettingsIcon size={16} /> Settings
          </button>
        </div>
      </aside>

      <main className="flex-1 min-w-0 overflow-y-auto">
        {error && (
          <div className="px-6 py-3 text-sm no-print flex items-center justify-between gap-3 flex-wrap" style={{ background: T.redTint, color: T.red, borderBottom: `1px solid ${T.red}` }}>
            <span className="font-semibold flex items-center gap-2"><AlertTriangle size={15} /> {error.message}</span>
            <div className="flex items-center gap-2">
              {error.retry && <Button variant="ghost" onClick={async () => { const fn = error.retry; setError(null); await fn(); }} style={{ borderColor: T.red, color: T.red, fontSize: 12, padding: "5px 10px" }}>Retry now</Button>}
              <button onClick={() => setError(null)} style={{ color: T.red }}><X size={16} /></button>
            </div>
          </div>
        )}
        {view === "home" && <HomeView index={index} records={records} role={role} orgName={settings.orgName} onGo={async (v) => { if (v !== "dashboard") await ensureAllLoaded(); setView(v); }} onNewEngagement={() => setShowNewEngagement(true)} onAskRadar={(q) => radarRef.current?.ask(q)} />}
        {view === "dashboard" && <Dashboard index={index} onOpen={openEngagement} onNew={() => setShowNewEngagement(true)} onImport={() => setShowImport(true)} records={records} role={role} />}
        {view === "actions" && <ActionTrackerView index={index} records={records} onOpen={openEngagement} orgName={settings.orgName} />}
        {view === "trends" && <TrendsView index={index} records={records} settings={settings} onUpdateThemes={updateThemes} />}
        {view === "detail" && activeId && <EngagementDetail record={records[activeId]} allRecords={records} onBack={() => setView("dashboard")} onUpdate={(updater) => updateRecord(activeId, updater)} settings={settings} onUpdateThemes={updateThemes} onClone={cloneEngagement} onDelete={deleteEngagement} onUpdateLibraries={updateControlLibraries} />}
      </main>

      {showNewEngagement && <NewEngagementModal onClose={() => setShowNewEngagement(false)} onCreate={createEngagement} index={index} />}
      {showImport && <ImportModal index={index} onClose={() => setShowImport(false)} onImport={importData} onImportMulti={importDataMulti} settings={settings} onUpdateThemes={updateThemes} />}
      <ChatWidget ref={radarRef} index={index} records={records} ensureAllLoaded={ensureAllLoaded} />
      <FeedbackWidget role={role} />
      {toast && (
        <div className="fixed flex items-center gap-2 px-4 py-2.5 rounded-full no-print" style={{ top: 20, left: "50%", transform: "translateX(-50%)", background: T.navy, color: "white", zIndex: 70, boxShadow: "0 6px 20px rgba(4,30,66,0.3)", fontSize: 13, fontWeight: 600 }}>
          <CheckCircle2 size={15} style={{ color: T.sky }} /> {toast}
        </div>
      )}
      {showSettings && (
        <SettingsModal
          settings={settings}
          onUpdateOrgName={updateOrgName}
          onUpdateAdminPin={updateAdminPin}
          onUpdateSharedLink={updateSharedLink}
          role={role}
          setRole={setRole}
          onManageThemes={() => setShowSettingsThemes(true)}
          onResetData={resetAllData}
          onClose={() => setShowSettings(false)}
        />
      )}
      {showSettingsThemes && <ThemeManager themes={settings.themes} onSave={updateThemes} onClose={() => setShowSettingsThemes(false)} />}
    </div>
  );
}

  return (
    <button onClick={onClick} className="flex items-center gap-2.5 px-3 py-2.5 rounded-md text-sm font-medium" style={{ background: active ? "rgba(98,181,229,0.15)" : "transparent", color: active ? T.sky : "#C9D6E8" }}>
      <Icon size={16} /> {label}
    </button>
  );
}

function DangerConfirmModal({ title, message, confirmWord, adminPin, onConfirm, onClose }) {
  const [typed, setTyped] = useState("");
  const [pin, setPin] = useState("");
  const [err, setErr] = useState("");
  const wordOk = typed.trim().toUpperCase() === confirmWord.toUpperCase();
  const submit = () => {
    if (!wordOk) { setErr(`Type "${confirmWord}" exactly to confirm.`); return; }
    if (adminPin && pin !== adminPin) { setErr("That admin PIN isn't correct."); return; }
    onConfirm();
  };
  return (
    <Modal title={title} onClose={onClose}>
      <div className="flex flex-col gap-3">
        <div className="rounded-lg p-3 flex items-start gap-2 text-sm" style={{ background: T.redTint, color: T.red }}>
          <AlertTriangle size={16} style={{ flexShrink: 0, marginTop: 1 }} /> {message}

function GaugeChart({ value, label, size = 150, thickness = 16 }) {
  const [mounted, setMounted] = useState(false);
  useEffect(() => { const t = setTimeout(() => setMounted(true), 30); return () => clearTimeout(t); }, []);
  const r = (size - thickness) / 2;
  const circ = 2 * Math.PI * r;
  const color = value >= 80 ? T.green : value >= 55 ? T.amber : T.red;
  const dash = mounted ? (value / 100) * circ : 0;
  return (
    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} style={{ filter: "drop-shadow(0 2px 6px rgba(4,30,66,0.12))" }}>
      <circle cx={size / 2} cy={size / 2} r={r} fill="none" stroke={T.grayTint} strokeWidth={thickness} />
      <circle
        cx={size / 2} cy={size / 2} r={r} fill="none" stroke={color} strokeWidth={thickness} strokeLinecap="round"
        strokeDasharray={`${dash} ${circ - dash}`} transform={`rotate(-90 ${size / 2} ${size / 2})`}
        style={{ transition: "stroke-dasharray 900ms cubic-bezier(.4,0,.2,1)" }}
      />
      <text x="50%" y="46%" textAnchor="middle" style={{ fontSize: size * 0.24, fontWeight: 800, fill: T.navy, fontFamily: "Inter, sans-serif" }}>{value}</text>
      <text x="50%" y="63%" textAnchor="middle" style={{ fontSize: 9, fontWeight: 600, fill: T.textMute, fontFamily: "Inter, sans-serif" }}>{label}</text>
    </svg>
  );
}

function ActionTrackerView({ index, records, onOpen, orgName }) {
  const [query, setQuery] = useState("");
  const [statusFilter, setStatusFilter] = useState("All");
  const [airportFilter, setAirportFilter] = useState("All");
  const [areaFilter, setAreaFilter] = useState("All");
  const [ownerFilter, setOwnerFilter] = useState("All");
  const [showSummary, setShowSummary] = useState(false);
  const [sentLog, setSentLog] = useState(null);

  useEffect(() => {
    (async () => {
      try {
        const res = await window.storage.get("assurance:summarySentLog", SHARED_STORAGE);
        setSentLog(res ? JSON.parse(res.value) : []);
      } catch (e) { setSentLog([]); }
    })();
  }, []);

  const rows = Object.values(records).flatMap((r) => r.actions.map((a) => ({ ...a, engTitle: r.meta.title, engId: r.meta.id, airport: r.meta.airport, area: r.meta.area })));
  const owners = Array.from(new Set(rows.map((a) => a.owner).filter(Boolean))).sort();
  const filtered = rows.filter((a) =>
    (airportFilter === "All" || a.airport === airportFilter) &&
    (areaFilter === "All" || a.area === areaFilter) &&
    (ownerFilter === "All" || a.owner === ownerFilter) &&
    (statusFilter === "All" || (statusFilter === "Overdue" ? isOverdue(a.dueDate, a.status) : a.status === statusFilter)) &&
    (a.description.toLowerCase().includes(query.toLowerCase()) || a.engTitle.toLowerCase().includes(query.toLowerCase()) || (a.owner || "").toLowerCase().includes(query.toLowerCase()))
  ).sort((a, b) => (a.dueDate || "9999") < (b.dueDate || "9999") ? -1 : 1);

  const overdueCount = rows.filter((a) => isOverdue(a.dueDate, a.status)).length;
  const hasActiveFilter = airportFilter !== "All" || areaFilter !== "All" || ownerFilter !== "All" || statusFilter !== "All";

  const summaryTitle = () => {
    const parts = [];
    parts.push(statusFilter === "Overdue" ? "Overdue actions" : statusFilter !== "All" ? `${statusFilter} actions` : "Actions");
    if (areaFilter !== "All") parts.push(`in ${areaFilter}`);
    if (airportFilter !== "All") parts.push(`at ${airportFilter}`);
    if (ownerFilter !== "All") parts.push(`for ${ownerFilter}`);
    return parts.join(" ");
  };

  const logSent = async (title) => {
    try {
      const res = await window.storage.get("assurance:summarySentLog", SHARED_STORAGE).catch(() => null);
      const list = res ? JSON.parse(res.value) : [];
      const next = [{ title, ts: new Date().toISOString() }, ...list.filter((l) => l.title !== title)].slice(0, 15);
      await window.storage.set("assurance:summarySentLog", JSON.stringify(next), SHARED_STORAGE);
      setSentLog(next);
    } catch (e) { /* best effort */ }
  };

  if (showSummary) {
    return <FilteredActionSummary title={summaryTitle()} actions={filtered} orgName={orgName} onSent={() => logSent(summaryTitle())} onBack={() => setShowSummary(false)} />;
  }

  return (
    <div className="p-6 md:p-8 max-w-6xl mx-auto">
      <div className="flex items-center justify-between flex-wrap gap-2">
        <div>
          <h1 className="text-2xl font-extrabold" style={{ color: T.navy }}>Action tracker</h1>
          <p className="text-sm mt-0.5 mb-5" style={{ color: T.textMute }}>Every action across every engagement — the single list for the weekly check-in.</p>
        </div>
        {hasActiveFilter && <Button variant="ghost" onClick={() => setShowSummary(true)}><Printer size={15} /> Generate summary — {summaryTitle()}</Button>}
      </div>

const STOP_WORDS = new Set(["a", "an", "and", "at", "for", "in", "of", "on", "the", "to", "with", "management", "review", "assurance"]);
function deriveControlPrefix(title) {
  const words = (title || "").split(/\s+/).map((w) => w.replace(/[^a-zA-Z]/g, "")).filter((w) => w && !STOP_WORDS.has(w.toLowerCase()));
  if (words.length === 0) return "GEN";
  if (words.length === 1) return words[0].slice(0, 3).toUpperCase();
  return words.slice(0, 4).map((w) => w[0]).join("").toUpperCase();
}
function nextControlRef(workbook, prefix) {
  const nums = (workbook || [])
    .filter((c) => c.ref && c.ref.toUpperCase().startsWith(prefix.toUpperCase()))
    .map((c) => parseInt((c.ref.match(/(\d+)$/) || [])[1], 10))
    .filter((n) => !isNaN(n));
  const next = nums.length > 0 ? Math.max(...nums) + 1 : 1;
  return `${prefix}${String(next).padStart(2, "0")}`;
}
function airportName(code) {
  return code === "MAN" ? "Manchester Airport" : code === "STN" ? "London Stansted Airport" : code === "EMA" ? "East Midlands Airport" : code === "CAVU" ? "CAVU (MAG's commercial arm)" : code;
}
function stripMarkdownNoise(text) {
  return text.replace(/\*\*/g, "").replace(/^#+\s*/gm, "").replace(/```/g, "");
}
function parseScopeDraft(rawText) {
  const text = stripMarkdownNoise(rawText);
  const grab = (label) => {
    const re = new RegExp(`${label}:\\s*([\\s\\S]*?)(?=\\n[A-Z ]+:|$)`, "i");
    const m = text.match(re);
    return m ? m[1].trim() : "";
  };
  const background = grab("BACKGROUND");
  const objective = grab("OBJECTIVE");
  const approach = grab("APPROACH");
  const limitations = grab("LIMITATIONS");
  const parseLines = (block) => block.split("\n").map((l) => l.replace(/^[-•]\s*/, "").trim()).filter(Boolean);
  const controls = parseLines(grab("SUGGESTED CONTROLS")).map((line, i) => {
    const parts = line.split(/\s*\|\s*/);
    const description = parts[0] || line;
    const testPart = parts.find((p) => /^test:/i.test(p));
    const refPart = parts.find((p) => /^ref:/i.test(p));
    return {
      id: uid("draft"), ref: `C${i + 1}`, description: description.trim(),
      test: testPart ? testPart.replace(/^test:\s*/i, "").trim() : "",
      regulatoryRef: refPart ? refPart.replace(/^ref:\s*/i, "").trim() : "",
      include: true,
    };
  });
  const evidence = parseLines(grab("SUGGESTED EVIDENCE REQUESTS")).map((line) => {
    const parts = line.split(/\s*\|\s*/);
    const description = parts[0] || line;
    const controlPart = parts.find((p) => /^control:/i.test(p));
    const idx = controlPart ? parseInt(controlPart.replace(/^control:\s*/i, "").trim(), 10) : null;
    const relatedControl = idx && controls[idx - 1] ? controls[idx - 1] : null;
    return { id: uid("draft"), description: description.trim(), controlDraftId: relatedControl ? relatedControl.id : null, include: true };
  });
  const sources = parseLines(grab("SOURCES"));
  return { background, objective, approach, limitations, controls, evidence, sources };
}

async function runAIScopeDraft(title, airportCode, area) {
  const prompt = `Research the current UK legal/regulatory framework and any applicable British Standards for a second-line assurance review titled "${title}" at ${airportName(airportCode)}, part of Manchester Airports Group, covering the area of ${area}. Search for the specific, current legislation and standards that apply.

Respond in EXACTLY this plain-text format, with these six headers in capitals, nothing before the first header:

BACKGROUND:
1-2 sentences on why this review is happening — the regulatory driver or programme context.

OBJECTIVE:
A 2-4 sentence assurance engagement objective and scope statement, referencing the specific legal/regulatory basis.

APPROACH:
1-2 sentences on how the review will be carried out — self-assessment, joint on-site review, evidence sampling.

LIMITATIONS:
1-2 sentences on typical exceptions and limitations for a review of this kind — reliance on management representations, sampling basis, what's out of scope.

SUGGESTED CONTROLS:
- <control statement to test> | Test: <how to test it> | Ref: <the specific regulation, clause or standard this exact control comes from>
- <control statement to test> | Test: <how to test it> | Ref: <the specific regulation, clause or standard this exact control comes from>
(5-8 controls, numbered 1 to N in the order you list them — you'll refer back to these numbers below)

SUGGESTED EVIDENCE REQUESTS:
- <specific evidence item to request> | Control: <the number, 1-N, of the control above that this evidence supports>
(4-6 items)

SOURCES:
- <name of regulation/standard>: <one-line description of what it requires>
(list every source actually used)`;
  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-6",
      max_tokens: 3000,
      messages: [{ role: "user", content: prompt }],
      tools: [{ type: "web_search_20250305", name: "web_search" }],
    }),
  });
  const data = await response.json();
  if (data.type === "error") throw new Error(data.error?.message || "Anthropic API error");
  const text = (data.content || []).filter((b) => b.type === "text").map((b) => b.text).join("\n");
  if (!text) throw new Error("No response text");
  const parsed = parseScopeDraft(text);
  if (!parsed.objective) throw new Error("Couldn't parse a scope from that response");
  return parsed;
}

  const [draft, setDraft] = useState(null); // { objective, controls, evidence, sources }
  const [aiDrafted, setAiDrafted] = useState(false);
  const set = (k, v) => setForm((f) => ({ ...f, [k]: v }));

  const draftScope = async () => {
    if (!form.title.trim()) { setTitleError(true); return; }
    setAiLoading(true);
    setAiError("");
    try {
      const parsed = await runAIScopeDraft(form.title, form.airport, form.area);
      setDraft(parsed);
      set("objective", parsed.objective);
      if (parsed.background) set("background", parsed.background);
      if (parsed.approach) set("approach", parsed.approach);
      if (parsed.limitations) set("limitations", parsed.limitations);
      setAiDrafted(true);
    } catch (err) {
      console.error("Scope drafter failed:", err);
      setAiError(err.message === "No response text" || err.message?.includes("parse") ? "Couldn't parse a scope from that response — try again, or write it manually." : `Couldn't draft a scope just now (${err.message}) — please try again.`);
    } finally {
      setAiLoading(false);
    }
  };

  const toggleItem = (list, id) => (list === "controls" ? setDraft((d) => ({ ...d, controls: d.controls.map((c) => c.id === id ? { ...c, include: !c.include } : c) })) : setDraft((d) => ({ ...d, evidence: d.evidence.map((e) => e.id === id ? { ...e, include: !e.include } : e) })));

  const submit = () => {
    if (!form.title.trim()) { setTitleError(true); return; }
    const payload = { ...form };
    if (aiDrafted && draft) {
      payload._draftControls = draft.controls.filter((c) => c.include);
      payload._draftEvidence = draft.evidence.filter((e) => e.include);
      payload._aiDrafted = true;
    }
    onCreate(payload);
  };

  return (
    <Modal title="New engagement" onClose={onClose} wide>
      <div className="flex flex-col gap-3">
        <Field label="Engagement title">
          <TextInput value={form.title} onChange={(e) => { set("title", e.target.value); if (titleError) setTitleError(false); }} placeholder="e.g. Asbestos Management at MAG" style={titleError ? { borderColor: T.red } : undefined} />
          {titleError && <span className="text-xs font-semibold" style={{ color: T.red }}>Enter a title before creating the engagement.</span>}
        </Field>
        <div className="grid grid-cols-3 gap-3">
          <Field label="Airport"><Select value={form.airport} onChange={(e) => set("airport", e.target.value)}>{AIRPORTS.map((a) => <option key={a}>{a}</option>)}</Select></Field>
          <Field label="Area" style={{ gridColumn: "span 2" }}><Select value={form.area} onChange={(e) => set("area", e.target.value)}>{AREAS.map((a) => <option key={a}>{a}</option>)}</Select></Field>
        </div>
        <Field label="Type"><Select value={form.type} onChange={(e) => set("type", e.target.value)}>{ENG_TYPES.map((t) => <option key={t}>{t}</option>)}</Select></Field>
        <Field label="Assurance lead"><Select value={form.lead} onChange={(e) => set("lead", e.target.value)}>{LEADS.map((l) => <option key={l}>{l}</option>)}</Select></Field>
        <Field label="Priority"><Select value={form.priority} onChange={(e) => set("priority", e.target.value)}>{PRIORITIES.map((p) => <option key={p}>{p}</option>)}</Select></Field>
        {(() => {
          const oneYearAgo = new Date(); oneYearAgo.setFullYear(oneYearAgo.getFullYear() - 1);
          const priorReview = index.find((e) => e.lead === form.lead && e.area === form.area && e.startDate && new Date(e.startDate) > oneYearAgo);
          return priorReview ? (
            <p className="text-xs font-semibold px-3 py-2 rounded" style={{ color: T.amber, background: T.amberTint }}>
              Independence check: {form.lead} already reviewed {form.area} within the last 12 months — "{priorReview.title}" ({fmtDate(priorReview.startDate)}). Worth confirming independence before assigning this one too.
            </p>
          ) : null;
        })()}
        <div className="grid grid-cols-2 gap-3">
          <Field label="Start date"><TextInput type="date" value={form.startDate} onChange={(e) => set("startDate", e.target.value)} /></Field>
          <Field label="Target completion"><TextInput type="date" value={form.targetDate} onChange={(e) => set("targetDate", e.target.value)} /></Field>
        </div>

        <div className="flex items-center justify-between">
          <span className="text-sm font-semibold" style={{ color: T.navy }}>Objective / scope</span>
          <Button variant="ghost" onClick={draftScope} style={{ fontSize: 12, padding: "5px 10px" }}>{aiLoading ? "Researching…" : <><Sparkles size={13} /> Draft scope with AI</>}</Button>
        </div>
        <TextArea rows={3} value={form.objective} onChange={(e) => { set("objective", e.target.value); setAiDrafted(false); }} placeholder="What is this engagement assuring, and over what scope? Or click 'Draft scope with AI' above." />
        {aiError && <p className="text-xs font-semibold" style={{ color: T.red }}>{aiError}</p>}

        {draft && aiDrafted && (
          <div className="rounded-lg p-3.5 flex flex-col gap-3" style={{ background: T.skyTint, border: `1px solid ${T.sky}` }}>
            <p className="text-xs font-semibold flex items-center gap-1.5" style={{ color: T.navy }}><Sparkles size={13} /> AI-drafted from a live web search — verify every source before relying on this. Untick anything you don't want.</p>
            {draft.sources.length > 0 && (
              <div>
                <div className="text-xs font-bold mb-1" style={{ color: T.navy }}>Sources found</div>
                <ul className="text-xs flex flex-col gap-0.5" style={{ color: T.text }}>{draft.sources.map((s, i) => <li key={i}>• {s}</li>)}</ul>
              </div>
            )}
            {draft.controls.length > 0 && (
              <div>
                <div className="text-xs font-bold mb-1" style={{ color: T.navy }}>Suggested controls to test ({draft.controls.filter((c) => c.include).length}/{draft.controls.length})</div>
                <div className="flex flex-col gap-1">
                  {draft.controls.map((c) => (
                    <label key={c.id} className="flex items-start gap-2 text-xs" style={{ color: T.text }}>
                      <input type="checkbox" checked={c.include} onChange={() => toggleItem("controls", c.id)} style={{ marginTop: 2 }} />
                      <span>{c.description}{c.test ? ` — test: ${c.test}` : ""}</span>
                    </label>
                  ))}
                </div>
              </div>
            )}
            {draft.evidence.length > 0 && (
              <div>
                <div className="text-xs font-bold mb-1" style={{ color: T.navy }}>Suggested evidence requests ({draft.evidence.filter((e) => e.include).length}/{draft.evidence.length})</div>
                <div className="flex flex-col gap-1">
                  {draft.evidence.map((e) => (
                    <label key={e.id} className="flex items-start gap-2 text-xs" style={{ color: T.text }}>
                      <input type="checkbox" checked={e.include} onChange={() => toggleItem("evidence", e.id)} style={{ marginTop: 2 }} />
                      <span>{e.description}</span>
                    </label>
                  ))}
                </div>
              </div>
            )}
          </div>
        )}

        <div className="flex justify-end gap-2 mt-2"><Button variant="ghost" onClick={onClose}>Cancel</Button><Button onClick={submit}>Create engagement</Button></div>
      </div>
    </Modal>
  );
}

const RISK_ALIASES = { "no priority": "Low", low: "Low", medium: "Medium", high: "High", critical: "Critical", urgent: "Critical" };
function normalizeRisk(raw) {
  const k = String(raw || "").trim().toLowerCase();
  return RISK_ALIASES[k] || "Medium";
}
function normalizeCompliance(raw) {
  const k = String(raw || "").trim().toLowerCase();
  if (!k) return "";
  if (k === "sd" || k.includes("serious")) return "Serious Deficiency";
  if (k.includes("major")) return "Non-Compliant – Major";
  if (k.includes("minor")) return "Non-Compliant – Minor";
  if (k.includes("non-compliance") || k.includes("non compliance")) return "Non-Compliant – Minor";
  if (k.includes("compliant") || k.includes("compliance")) return "Control Test Compliant";
  return "";
}
function normalizeActionStatus(raw) {
  const k = String(raw || "").trim().toLowerCase();
  if (k.includes("complete") || k.includes("closed")) return "Closed";
  if (k.includes("progress") || k.includes("timeframe")) return "In Progress";
  return "Open";
}
const LEAD_ALIASES = { "becky bassett": "Rebecca Bassett" };
function normalizeLead(raw) {
  const name = String(raw || "").trim();
  return LEAD_ALIASES[name.toLowerCase()] || name;
}
function normalizeArea(raw) {
  const k = String(raw || "").trim();
  if (!k) return "";
  const match = AREAS.find((a) => a.toLowerCase() === k.toLowerCase());
  if (match) return match;
  if (k.toLowerCase().includes("ops resilience") || k.toLowerCase().includes("operational")) return "Operational Resilience";
  return k;
}
function excelDateToISO(raw) {
  if (!raw) return "";
  if (raw instanceof Date) return raw.toISOString().slice(0, 10);
  const s = String(raw).trim();
  const d = new Date(s);
  return isNaN(d) ? "" : d.toISOString().slice(0, 10);
}

function ImportModal({ index, onClose, onImport, onImportMulti, settings, onUpdateThemes }) {
  const [step, setStep] = useState(1);
  const [mode, setMode] = useState("multi");
  const [fileName, setFileName] = useState("");
  const [headers, setHeaders] = useState([]);
  const [rows, setRows] = useState([]);
  const [mapping, setMapping] = useState({});
  const [target, setTarget] = useState(index[0]?.id || "__new__");
  const [newTitle, setNewTitle] = useState("Imported tracker data");
  const [error, setError] = useState("");
  const [preview, setPreview] = useState(null);

  const handleFile = async (e) => {
    const file = e.target.files[0];
    if (!file) return;
    setError("");
    setFileName(file.name);
    try {
      const buf = await file.arrayBuffer();
      const wb = XLSX.read(buf, { type: "array", cellDates: true });
      const sheetName = wb.SheetNames.find((n) => /action|tracker|finding/i.test(n)) || wb.SheetNames[0];
      const sheet = wb.Sheets[sheetName];
      const json = XLSX.utils.sheet_to_json(sheet, { defval: "" });
      if (json.length === 0) { setError("No rows found on that sheet."); return; }
      const hdrs = Object.keys(json[0]);
      const guessed = {};
      Object.entries(IMPORT_GUESS).forEach(([key, kws]) => {
        const match = hdrs.find((h) => kws.some((k) => h.toLowerCase().includes(k)));
        guessed[key] = match || "";
      });
      setHeaders(hdrs);
      setRows(json);
      setMapping(guessed);
      setMode(guessed.reviewTitle && guessed.airport ? "multi" : "single");
      setStep(2);
    } catch (err) {
      setError("Could not read that file. Try exporting it as .xlsx or .csv first.");
    }
  };

  const buildFinding = (r) => {
    const ref = mapping.findingRef ? String(r[mapping.findingRef] || "") : "";
    const theme = mapping.theme ? String(r[mapping.theme] || "") : "";
    const compliance = mapping.complianceResult ? normalizeCompliance(r[mapping.complianceResult]) : "";
    let title = mapping.findingTitle ? String(r[mapping.findingTitle] || "") : "";
    if (title.length > 140) title = title.slice(0, 137) + "…";
    if (!title) title = [ref, theme].filter(Boolean).join(" – ") || "Untitled";
    const rawDescription = mapping.description ? String(r[mapping.description] || "") : "";
    return {
      id: uid("fnd"),
      ref,
      title,
      description: compliance ? `[${compliance}] ${rawDescription}` : rawDescription,
      riskRating: mapping.riskRating ? normalizeRisk(r[mapping.riskRating]) : "Medium",
      theme,
      dateRaised: mapping.reviewDate ? (excelDateToISO(r[mapping.reviewDate]) || todayISO()) : todayISO(),
      _actionDescription: mapping.actionDescription ? String(r[mapping.actionDescription] || "") : "",
      _actionOwner: mapping.actionOwner ? String(r[mapping.actionOwner] || "") : "",
      _actionDue: mapping.actionDue ? excelDateToISO(r[mapping.actionDue]) : "",
      _actionStatus: mapping.actionStatus ? normalizeActionStatus(r[mapping.actionStatus]) : "Open",
    };
  };

  const buildGroups = () => {
    const map = new Map();
    rows.forEach((r) => {
      const titleVal = mapping.reviewTitle ? String(r[mapping.reviewTitle] || "Untitled review") : "Untitled review";
      const airportVal = mapping.airport ? String(r[mapping.airport] || "") : "";
      const yearVal = mapping.auditYear ? String(r[mapping.auditYear] || "") : "";
      const key = `${titleVal}|${airportVal}|${yearVal}`;
      if (!map.has(key)) {
        map.set(key, {
          title: `${titleVal}${airportVal ? " – " + airportVal : ""}${yearVal ? " (" + yearVal + ")" : ""}`,
          airport: AIRPORTS.find((a) => airportVal.toLowerCase().startsWith(a.toLowerCase())) || (airportVal.toLowerCase().includes("stansted") ? "STN" : airportVal.toLowerCase().includes("manchester") ? "MAN" : airportVal.toLowerCase().includes("midland") ? "EMA" : "MAN"),
          area: mapping.area ? normalizeArea(r[mapping.area]) : "",
          lead: mapping.lead ? normalizeLead(r[mapping.lead]) : "",
          startDate: mapping.reviewDate ? excelDateToISO(r[mapping.reviewDate]) : "",
          findings: [],
        });
      }
      map.get(key).findings.push(buildFinding(r));
    });
    return Array.from(map.values());
  };

  const newThemesFound = () => {
    if (!mapping.theme) return [];
    const seen = new Set();
    rows.forEach((r) => { const t = String(r[mapping.theme] || "").trim(); if (t) seen.add(t); });
    return Array.from(seen).filter((t) => !settings.themes.includes(t));
  };

  const doPreview = () => {
    if (mode === "multi") {
      if (!mapping.reviewTitle) { setError("Map a review/engagement title column, or switch to single-engagement mode."); return; }
      const groups = buildGroups();
      setPreview({ groups });
    } else {
      if (!mapping.findingTitle) { setError("Map at least a finding title/control column before importing."); return; }
      setPreview({ findings: rows.map(buildFinding) });
    }
    setError("");
    setStep(3);
  };

  const confirmImport = () => {
    const newThemes = newThemesFound();
    if (newThemes.length > 0) onUpdateThemes([...settings.themes, ...newThemes]);
    if (mode === "multi") onImportMulti(preview.groups);
    else onImport({ targetEngagement: target, newTitle, findings: preview.findings });
  };

  const fieldSet = mode === "multi" ? [...IMPORT_FIELDS_GROUPING, ...IMPORT_FIELDS_FINDING] : IMPORT_FIELDS_FINDING;

  return (
    <Modal title="Import from Excel" onClose={onClose} wide>
      <p className="text-xs mb-3" style={{ color: T.textMute }}>Reads your tracker directly in the browser — nothing is uploaded anywhere else. Works with .xlsx, .xls or .csv (including a Planner export to Excel).</p>

      {step === 1 && (
        <div className="flex flex-col gap-3">
          <div className="rounded-lg border-2 border-dashed p-8 flex flex-col items-center gap-2 text-center" style={{ borderColor: T.border }}>
            <UploadCloud size={26} style={{ color: T.sky }} />
            <p className="text-sm font-semibold" style={{ color: T.navy }}>Choose your tracker file</p>
            <p className="text-xs" style={{ color: T.textMute }}>The most relevant sheet's column headers will be used for mapping.</p>
            <input type="file" accept=".xlsx,.xls,.csv" onChange={handleFile} className="mt-2 text-xs" />
          </div>
          {error && <p className="text-xs font-semibold" style={{ color: T.red }}>{error}</p>}
          <div className="flex justify-end"><Button variant="ghost" onClick={onClose}>Cancel</Button></div>
        </div>
      )}

      {step === 2 && (
        <div className="flex flex-col gap-4">
          <div className="flex items-center gap-2 text-xs" style={{ color: T.textMute }}><FileCheck2 size={14} /> {fileName} — {rows.length} row(s) detected</div>

          <div>
            <div className="text-sm font-bold mb-2" style={{ color: T.navy }}>Import style</div>
            <div className="flex gap-2">
              <FilterChip label="Split into separate engagements (recommended)" active={mode === "multi"} onClick={() => setMode("multi")} />
              <FilterChip label="Import into one engagement" active={mode === "single"} onClick={() => setMode("single")} />
            </div>
          </div>

          {mode === "single" && (
            <div>
              <div className="text-sm font-bold mb-2" style={{ color: T.navy }}>Import into</div>
              <Select value={target} onChange={(e) => setTarget(e.target.value)}>
                <option value="__new__">+ Create a new engagement for this import</option>
                {index.map((e) => <option key={e.id} value={e.id}>{e.title}</option>)}
              </Select>
              {target === "__new__" && <TextInput value={newTitle} onChange={(e) => setNewTitle(e.target.value)} placeholder="Engagement title" style={{ marginTop: 8, width: "100%" }} />}
            </div>
          )}

          <div>
            <div className="text-sm font-bold mb-2" style={{ color: T.navy }}>Map your columns</div>
            <div className="grid grid-cols-2 gap-3">
              {fieldSet.map((f) => (
                <Field key={f.key} label={f.label + (f.required ? " *" : "")}>
                  <Select value={mapping[f.key] || ""} onChange={(e) => setMapping((m) => ({ ...m, [f.key]: e.target.value }))}>
                    <option value="">— not in file —</option>
                    {headers.map((h) => <option key={h} value={h}>{h}</option>)}
                  </Select>
                </Field>
              ))}
            </div>
          </div>

          {error && <p className="text-xs font-semibold" style={{ color: T.red }}>{error}</p>}
          <div className="flex justify-end gap-2">
            <Button variant="ghost" onClick={onClose}>Cancel</Button>
            <Button onClick={doPreview}>Preview import</Button>
          </div>
        </div>
      )}

      {step === 3 && preview && (
        <div className="flex flex-col gap-4">
          {mode === "multi" ? (
            <>
              <div className="text-sm font-bold" style={{ color: T.navy }}>{preview.groups.length} engagement(s) will be created</div>
              <div className="max-h-64 overflow-y-auto rounded-lg" style={{ border: `1px solid ${T.border}` }}>
                <Table cols={["Engagement", "Airport", "Area", "Lead", "Findings"]} rows={preview.groups.slice(0, 25).map((g) => [g.title, g.airport, g.area || "—", g.lead || "—", g.findings.length])} />
              </div>
              {preview.groups.length > 25 && <p className="text-xs" style={{ color: T.textMute }}>…and {preview.groups.length - 25} more.</p>}
            </>
          ) : (
            <>
              <div className="text-sm font-bold" style={{ color: T.navy }}>{preview.findings.length} finding(s) will be imported</div>
              <Table cols={["Finding", "Risk", "Action"]} rows={preview.findings.slice(0, 5).map((f) => [f.title, f.riskRating, f._actionDescription || "—"])} />
            </>
          )}
          {newThemesFound().length > 0 && <p className="text-xs" style={{ color: T.textMute }}>{newThemesFound().length} new theme(s) from your data will be added to the theme list: {newThemesFound().slice(0, 6).join(", ")}{newThemesFound().length > 6 ? "…" : ""}</p>}
          <div className="flex justify-end gap-2">
            <Button variant="ghost" onClick={() => setStep(2)}>Back</Button>
            <Button onClick={confirmImport}>Confirm import</Button>
          </div>
        </div>
      )}
    </Modal>
  );
}

function RiskHeatMap({ findings }) {
  const severe = (r) => r === "Critical" || r === "High";
  const cell = (airport, area) => findings.filter((f) => f.airport === airport && f.area === area);
  const heatColor = (n, sevN) => {
    if (n === 0) return T.grayTint;
    if (sevN > 0) return "#F1B8B0";
    if (n >= 3) return T.amberTint;
    return T.greenTint;
  };
  return (
    <div className="rounded-lg p-5" style={{ background: T.card, border: `1px solid ${T.border}` }}>
      <div className="flex items-center gap-2 mb-1">
        <div className="flex items-center justify-center rounded-md flex-shrink-0" style={{ width: 26, height: 26, background: T.redTint }}><AlertTriangle size={14} style={{ color: T.red }} /></div>
        <h4 className="text-sm font-bold" style={{ color: T.navy }}>Risk heat-map</h4>
      </div>
      <p className="text-xs mb-4" style={{ color: T.textMute }}>Findings by airport and area — darker cells carry Critical/High risk findings.</p>
      <div className="overflow-x-auto">
        <table className="text-xs" style={{ borderCollapse: "separate", borderSpacing: 4 }}>
          <thead>
            <tr>
              <th></th>
              {AIRPORTS.map((a) => <th key={a} className="font-semibold px-2 pb-1" style={{ color: T.navy }}>{a}</th>)}
            </tr>
          </thead>
          <tbody>
            {AREAS.map((area) => (
              <tr key={area}>
                <td className="font-semibold pr-3 whitespace-nowrap text-right" style={{ color: T.textMute }}>{area}</td>
                {AIRPORTS.map((airport) => {
                  const items = cell(airport, area);
                  const sevN = items.filter((f) => severe(f.riskRating)).length;
                  return (
                    <td key={airport}>
                      <div className="rounded-md flex items-center justify-center font-bold" style={{ width: 56, height: 40, background: heatColor(items.length, sevN), color: T.navy }} title={`${items.length} finding(s)`}>
                        {items.length > 0 ? items.length : "—"}
                      </div>
                    </td>
                  );
                })}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

function TrendsView({ index, records, settings, onUpdateThemes }) {

  const [period, setPeriod] = useState("All time");

  const engFilter = (e) => (airportFilter === "All" || e.airport === airportFilter) && (areaFilter === "All" || e.area === areaFilter) && (leadFilter === "All" || e.lead === leadFilter);
  const filteredIndex = index.filter(engFilter);
  const filteredIds = new Set(filteredIndex.map((e) => e.id));
  const filteredRecords = Object.values(records).filter((r) => filteredIds.has(r.meta.id));

  const range = periodRange(period);
  const prevRange = period === "All time" ? null : periodRange(period, 1);

  const allFindingsRaw = filteredRecords.flatMap((r) => r.findings.map((f) => ({ ...f, airport: r.meta.airport, area: r.meta.area, lead: r.meta.lead })));
  const allFindings = allFindingsRaw.filter((f) => inRange(f.dateRaised, range));
  const prevFindings = prevRange ? allFindingsRaw.filter((f) => inRange(f.dateRaised, prevRange)) : [];
  const allActions = filteredRecords.flatMap((r) => r.actions.map((a) => ({ ...a, airport: r.meta.airport, area: r.meta.area })));
  const allControls = filteredRecords.flatMap((r) => r.workbook);

  const countBy = (arr, key) => {
    const map = {};
    arr.forEach((x) => { const k = x[key] || "Untagged"; map[k] = (map[k] || 0) + 1; });
    return Object.entries(map).sort((a, b) => b[1] - a[1]);
  };
  const byTheme = countBy(allFindings, "theme");
  const byAirport = countBy(allFindings, "airport");
  const byArea = countBy(allFindings, "area");
  const byRisk = countBy(allFindings, "riskRating");
  const byLead = countBy(allFindings, "lead");
  const overdueActions = allActions.filter((a) => isOverdue(a.dueDate, a.status));
  const overdueByAirport = countBy(overdueActions, "airport");
  const recommendationsCount = allFindings.filter((f) => (f.findingType || "Recommendation") === "Recommendation").length;
  const improvementOppsCount = allFindings.filter((f) => f.findingType === "Improvement Opportunity").length;

  const maxOf = (arr) => Math.max(1, ...arr.map(([, v]) => v));

  const stageSegments = STAGES.map((s) => ({ label: s, value: filteredIndex.filter((e) => e.stage === s).length, color: STAGE_COLORS[s] }));
  const riskSegments = RISK_RATINGS.map((r) => ({ label: r, value: allFindings.filter((f) => f.riskRating === r).length, color: riskColor(r).c }));
  const controlSegments = CONTROL_RESULTS.map((r) => ({ label: r, value: allControls.filter((c) => c.result === r).length, color: statusColor(r).c }));
  const actionSegments = [
    { label: "Closed", value: allActions.filter((a) => a.status === "Closed").length, color: T.green },
    { label: "Overdue", value: overdueActions.length, color: T.red },
    { label: "In Progress", value: allActions.filter((a) => a.status === "In Progress" && !isOverdue(a.dueDate, a.status)).length, color: T.amber },
    { label: "Open", value: allActions.filter((a) => a.status === "Open" && !isOverdue(a.dueDate, a.status)).length, color: T.gray },
  ];

  const trendsData = { filteredIndex, allFindings, allActions, allControls, byTheme, byAirport, byArea, byRisk, byLead, overdueByAirport, overdueActions, stageSegments, riskSegments, controlSegments, actionSegments, maxOf, recommendationsCount, improvementOppsCount, filters: { airportFilter, areaFilter, leadFilter, period } };

  if (showReport) return <TrendsReportView data={trendsData} orgName={settings.orgName} onBack={() => setShowReport(false)} />;

  return (
    <div className="p-6 md:p-8 max-w-6xl mx-auto">
      <div className="flex items-center justify-between mb-1 flex-wrap gap-3">
        <div>
          <h1 className="text-2xl font-extrabold" style={{ color: T.navy }}>Trends & themes</h1>
          <p className="text-sm mt-0.5" style={{ color: T.textMute }}>Findings and actions rolled up across engagements, for PMO reporting to Conor and the Group Safety Resilience Board.</p>
        </div>
        <div className="flex items-center gap-2">
          <Button variant="ghost" onClick={() => setShowThemeManager(true)}><Settings2 size={15} /> Manage themes</Button>
          <Button onClick={() => setShowReport(true)}><Printer size={15} /> Generate report</Button>
        </div>
      </div>

      <div className="flex flex-col gap-2 mt-5 mb-2">
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-xs font-semibold w-16 flex-shrink-0" style={{ color: T.textMute }}>Airport</span>
          <FilterChip label="All" active={airportFilter === "All"} onClick={() => setAirportFilter("All")} />
          {AIRPORTS.map((a) => <FilterChip key={a} label={a} active={airportFilter === a} onClick={() => setAirportFilter(a)} />)}
        </div>
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-xs font-semibold w-16 flex-shrink-0" style={{ color: T.textMute }}>Area</span>
          <FilterChip label="All" active={areaFilter === "All"} onClick={() => setAreaFilter("All")} />
          {AREAS.map((a) => <FilterChip key={a} label={a} active={areaFilter === a} onClick={() => setAreaFilter(a)} />)}
        </div>
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-xs font-semibold w-16 flex-shrink-0" style={{ color: T.textMute }}>Lead</span>
          <FilterChip label="All" active={leadFilter === "All"} onClick={() => setLeadFilter("All")} />
          {LEADS.map((l) => <FilterChip key={l} label={l} active={leadFilter === l} onClick={() => setLeadFilter(l)} />)}
        </div>
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-xs font-semibold w-16 flex-shrink-0" style={{ color: T.textMute }}>Period</span>
          {PERIODS.map((p) => <FilterChip key={p} label={p} active={period === p} onClick={() => setPeriod(p)} />)}
        </div>
      </div>

      <div className="grid md:grid-cols-3 gap-4 mt-7 mb-4">
        <Panel title="Engagement pipeline" icon={Compass} accent={T.ocean} accentBg={T.skyTint}><RingChart segments={stageSegments} size={104} thickness={13} centerLabel="engagements" /></Panel>
        <Panel title="Control testing outcomes" icon={ListChecks} accent={T.teal} accentBg={T.tealTint}>{allControls.length === 0 ? <p className="text-xs" style={{ color: T.textMute }}>No controls tested yet.</p> : <RingChart segments={controlSegments} size={104} thickness={13} centerLabel="controls" />}</Panel>
        <Panel title="Action status" icon={ListTodo} accent={T.navy} accentBg={T.grayTint}>{allActions.length === 0 ? <p className="text-xs" style={{ color: T.textMute }}>No actions raised yet.</p> : <RingChart segments={actionSegments} size={104} thickness={13} centerLabel="actions" />}</Panel>
      </div>

      {period !== "All time" && (
        <div className="flex items-center gap-2 mb-4 flex-wrap">
          <span className="text-xs font-semibold" style={{ color: T.textMute }}>Findings this period: {allFindings.length}</span>
          <Delta current={allFindings.length} previous={prevFindings.length} />
        </div>
      )}

      {allFindings.length === 0 ? (
        <div className="mt-4"><EmptyState icon={TrendingUp} text="No findings match these filters yet." /></div>
      ) : (
        <div className="grid md:grid-cols-2 gap-4">
          <Panel title="Findings by theme" icon={Sparkles} accent={T.ocean} accentBg={T.skyTint}>{byTheme.map(([k, v]) => <BarRow key={k} label={k} value={v} max={maxOf(byTheme)} color={T.ocean} />)}</Panel>
          <Panel title="Recommendations vs improvement opportunities" icon={ListTodo} accent={T.teal} accentBg={T.tealTint}>
            <div className="grid grid-cols-2 gap-3">
              <div className="rounded-lg p-3 text-center" style={{ background: T.grayTint }}>
                <div className="text-2xl font-extrabold" style={{ color: T.navy }}>{recommendationsCount}</div>
                <div className="text-xs mt-1" style={{ color: T.textMute }}>Recommendations issued</div>
                <div className="text-xs" style={{ color: T.textMute }}>tracked to completion</div>
              </div>
              <div className="rounded-lg p-3 text-center" style={{ background: T.tealTint }}>
                <div className="text-2xl font-extrabold" style={{ color: T.teal }}>{improvementOppsCount}</div>
                <div className="text-xs mt-1" style={{ color: T.textMute }}>Improvement opportunities</div>
                <div className="text-xs" style={{ color: T.textMute }}>logged, no owner/deadline</div>
              </div>
            </div>
          </Panel>
          <Panel title="Findings by risk rating" icon={AlertTriangle} accent={T.red} accentBg={T.redTint}><RingChart segments={riskSegments} size={92} thickness={12} centerLabel="findings" /></Panel>
          <Panel title="Findings by airport" icon={Compass} accent={T.sky} accentBg={T.skyTint}>{byAirport.map(([k, v]) => <BarRow key={k} label={k} value={v} max={maxOf(byAirport)} color={T.sky} />)}</Panel>
          <Panel title="Findings by area" icon={ClipboardList} accent={T.navy} accentBg={T.grayTint}>{byArea.map(([k, v]) => <BarRow key={k} label={k} value={v} max={maxOf(byArea)} color={T.navy} />)}</Panel>
          <Panel title="Findings by assurance lead" icon={Users} accent={T.teal} accentBg={T.tealTint}>{byLead.map(([k, v]) => <BarRow key={k} label={k} value={v} max={maxOf(byLead)} color={T.teal} />)}</Panel>
          <Panel title="Overdue actions by airport" icon={AlertTriangle} accent={T.red} accentBg={T.redTint}>
            {overdueByAirport.length === 0 ? <p className="text-xs" style={{ color: T.textMute }}>No overdue actions — nothing currently breaching due dates.</p> : overdueByAirport.map(([k, v]) => <BarRow key={k} label={k} value={v} max={maxOf(overdueByAirport)} color={T.red} />)}
          </Panel>
        </div>
      )}

      {allFindings.length > 0 && (
        <div className="mt-4">
          <RiskHeatMap findings={allFindings} />
        </div>
      )}

      <p className="text-xs mt-6" style={{ color: T.textMute }}>Findings are filtered by date raised; engagement and action charts reflect the airport/area/lead filters only. Period comparisons need "date raised" set on findings to be meaningful.</p>

      {showThemeManager && <ThemeManager themes={settings.themes} onSave={onUpdateThemes} onClose={() => setShowThemeManager(false)} />}
    </div>
  );
}
function Panel({ title, children, full, icon: Icon, accent, accentBg }) {
  return (
    <div className="rounded-lg p-5" style={{ background: T.card, border: `1px solid ${T.border}`, borderTop: accent ? `3px solid ${accent}` : undefined, gridColumn: full ? "1 / -1" : undefined }}>
      <div className="flex items-center gap-2 mb-3">
        {Icon && (
          <div className="flex items-center justify-center rounded-md flex-shrink-0" style={{ width: 26, height: 26, background: accentBg || T.skyTint }}>
            <Icon size={14} style={{ color: accent || T.ocean }} />
          </div>
        )}
        <h4 className="text-sm font-bold" style={{ color: T.navy }}>{title}</h4>
      </div>
      {children}
    </div>
  );
}
function ThemeManager({ themes, onSave, onClose }) {
  const [list, setList] = useState(themes);
  const [newTheme, setNewTheme] = useState("");
  const add = () => { if (!newTheme.trim() || list.includes(newTheme.trim())) return; setList([...list, newTheme.trim()]); setNewTheme(""); };
  const remove = (t) => { if (window.confirm(`Remove the theme "${t}"? This only affects the list — findings already tagged with it keep the label.`)) setList(list.filter((x) => x !== t)); };
  return (
    <Modal title="Manage assurance themes" onClose={onClose}>
      <p className="text-xs mb-3" style={{ color: T.textMute }}>These appear as a dropdown when tagging findings, so trends can be built by theme over time.</p>
      <div className="flex flex-col gap-2 mb-3">
        {list.map((t) => (
          <div key={t} className="flex items-center justify-between px-3 py-2 rounded" style={{ background: T.bg }}>
            <span className="text-sm" style={{ color: T.text }}>{t}</span>
            <button onClick={() => remove(t)} style={{ color: T.red, fontSize: 12, fontWeight: 600 }}>Remove</button>
          </div>
        ))}
      </div>
      <div className="flex gap-2 mb-4">
        <TextInput placeholder="Add a new theme…" value={newTheme} onChange={(e) => setNewTheme(e.target.value)} style={{ flex: 1 }} />
        <Button variant="ghost" onClick={add}><Plus size={14} /> Add</Button>
      </div>
      <div className="flex justify-end gap-2"><Button variant="ghost" onClick={onClose}>Cancel</Button><Button onClick={() => { onSave(list); onClose(); }}>Save themes</Button></div>
    </Modal>
  );
}

function SignOffPanel({ record, onUpdate }) {
  const { meta } = record;
  const status = meta.approvalStatus || "Draft";
  const [approver, setApprover] = useState(meta.approver || "");
  const [comments, setComments] = useState(meta.approvalComments || "");
  const sameAsLead = approver.trim() && meta.lead && approver.trim().toLowerCase() === meta.lead.trim().toLowerCase();

  const approve = () => {
    if (!approver.trim() || sameAsLead) return;
    onUpdate((r) => ({
      ...r,
      meta: { ...r.meta, approvalStatus: "Approved", approver: approver.trim(), approvalDate: todayISO(), approvalComments: comments },

/* ---- Programme milestones — the "where is this programme at" view ---- */
function milestoneStatus(m) {
  if (m.completedDate) return "Done";
  if (m.targetDate && m.targetDate < todayISO()) return "Overdue";
  if (m.targetDate) return "Scheduled";
  return "Not started";
}
function milestoneColor(status) {
  if (status === "Done") return { c: T.green, bg: T.greenTint };
  if (status === "Overdue") return { c: T.red, bg: T.redTint };
  if (status === "Scheduled") return { c: T.ocean, bg: T.skyTint };
  return { c: T.gray, bg: T.grayTint };
}
function MilestonesTab({ record, onUpdate }) {
  const milestones = record.milestones || [];
  const setField = (id, field, value) => onUpdate((r) => ({ ...r, milestones: (r.milestones || []).map((m) => m.id === id ? { ...m, [field]: value } : m) }));
  const toggleDone = (m) => setField(m.id, "completedDate", m.completedDate ? "" : todayISO());
  const addMilestone = () => onUpdate((r) => ({ ...r, milestones: [...(r.milestones || []), { id: uid("ms"), name: "New milestone", targetDate: "", completedDate: "" }] }));
  const removeMilestone = (id) => { if (window.confirm("Remove this milestone?")) onUpdate((r) => ({ ...r, milestones: (r.milestones || []).filter((m) => m.id !== id) })); };
  const initStandard = () => onUpdate((r) => ({ ...r, milestones: defaultMilestones() }));

  if (milestones.length === 0) {
    return <EmptyState icon={ListTodo} text="No programme milestones set up for this engagement yet." action={<Button onClick={initStandard}><Plus size={15} /> Add standard milestones</Button>} />;
  }

  const done = milestones.filter((m) => m.completedDate).length;

  return (
    <div>
      <div className="flex items-center justify-between mb-4">
        <p className="text-xs" style={{ color: T.textMute }}>{done} of {milestones.length} steps complete — mirrors the weekly programme-status conversation.</p>
        <Button variant="ghost" onClick={addMilestone}><Plus size={14} /> Add milestone</Button>
      </div>
      <div className="flex flex-col gap-2">
        {milestones.map((m, i) => {
          const status = milestoneStatus(m);
          const col = milestoneColor(status);
          return (
            <div key={m.id} className="flex items-center gap-3 rounded-lg p-3 flex-wrap" style={{ background: T.card, border: `1px solid ${T.border}` }}>
              <button onClick={() => toggleDone(m)} className="flex items-center justify-center rounded-full flex-shrink-0" style={{ width: 24, height: 24, background: m.completedDate ? T.green : T.grayTint, color: m.completedDate ? "white" : T.textMute }}>
                {m.completedDate ? <CheckCircle2 size={15} /> : <span className="text-xs font-bold">{i + 1}</span>}
              </button>
              <TextInput value={m.name} onChange={(e) => setField(m.id, "name", e.target.value)} style={{ flex: 1, minWidth: 180, fontWeight: 600 }} />
              <div className="flex items-center gap-1.5 text-xs" style={{ color: T.textMute }}>
                Target <TextInput type="date" value={m.targetDate} onChange={(e) => setField(m.id, "targetDate", e.target.value)} style={{ padding: "5px 7px", fontSize: 12 }} />
              </div>
              <div className="flex items-center gap-1.5 text-xs" style={{ color: T.textMute }}>
                Completed <TextInput type="date" value={m.completedDate} onChange={(e) => setField(m.id, "completedDate", e.target.value)} style={{ padding: "5px 7px", fontSize: 12 }} />
              </div>
              <Pill text={status} color={col} />
              <button onClick={() => removeMilestone(m.id)} style={{ color: T.red, fontSize: 11, fontWeight: 600 }}>×</button>

  };

  const copyText = async () => {

    try { await navigator.clipboard.writeText(text); setCopied(true); setTimeout(() => setCopied(false), 2000); markIssued(); } catch (e) { /* clipboard unavailable */ }

                    <Pill text={i.selfRating} color={statusColor(i.selfRating)} />

      )}

const EVIDENCE_TYPES = ["Digital link (SharePoint/Teams/URL)", "Physical or other (describe location)"];

function parseAIReview(rawText) {
  const text = stripMarkdownNoise(rawText);
  const grab = (label) => {
    const re = new RegExp(`${label}:\\s*([\\s\\S]*?)(?=\\n[A-Z ]+:|$)`, "i");
    const m = text.match(re);
    return m ? m[1].trim() : "";
  };
  return { assessment: grab("ASSESSMENT"), reasoning: grab("REASONING"), gaps: grab("GAPS") };
}

        body: JSON.stringify({ model: "claude-sonnet-4-6", max_tokens: 500, messages: [{ role: "user", content: prompt }] }),
      });
      const data = await response.json();

                    </div>
                    <div className="text-sm font-semibold" style={{ color: T.text }}>{e.description}</div>

                    <Select value={e.status} onChange={(ev) => setStatus(e.id, ev.target.value)} style={{ fontSize: 12, padding: "5px 8px" }}>{EVIDENCE_STATUSES.map((s) => <option key={s}>{s}</option>)}</Select>

                    <p className="text-xs mt-1.5 mb-1.5" style={{ color: T.textMute }}>There's no file upload here — this system can't store or open documents, and a SharePoint link can't be read either. Copy the text out of the document (Ctrl/Cmd+A, Ctrl/Cmd+C in the original file) and paste it below — the AI review then reads the actual content, not a summary of it.</p>

                    </div>
                  )}

              </div>
            );
          })}
        </div>
      )}
      {showForm && <EvidenceForm onClose={() => setShowForm(false)} onSave={add} controls={record.workbook} />}

  );
}

                      <Pill text={f.riskRating} color={riskColor(f.riskRating)} />
                      {(f.findingType || "Recommendation") === "Improvement Opportunity" && <Pill text="Improvement opportunity" color={{ c: T.teal, bg: T.tealTint }} />}
                      {f.theme && <Pill text={f.theme} color={{ c: T.ocean, bg: T.skyTint }} />}
                    </div>
                    <div className="text-sm font-bold" style={{ color: T.text }}>{f.title}</div>
                    <div className="text-xs mt-1" style={{ color: T.textMute }}>{f.description}</div>
                    {f.rootCause && <div className="text-xs mt-1 italic" style={{ color: T.textMute }}>Root cause: {f.rootCause}</div>}
                  </div>
                  <div className="flex items-center gap-2">
                    {(f.findingType || "Recommendation") !== "Improvement Opportunity" && <Button variant="ghost" onClick={() => setActionFormFor(f.id)}><Plus size={13} /> Action</Button>}
                    <button onClick={() => setEditingFinding(f)} style={{ color: T.ocean }}><Pencil size={13} /></button>
                    <button onClick={() => removeFinding(f.id)} style={{ color: T.red, fontSize: 12, fontWeight: 600 }}>Remove</button>
                  </div>
                </div>
                {acts.length > 0 && (
                  <div className="mt-3 pt-3 flex flex-col gap-2" style={{ borderTop: `1px solid ${T.border}` }}>
                    {acts.map((a) => {
                      const overdue = isOverdue(a.dueDate, a.status);
                      return (
                        <div key={a.id} className="flex items-center justify-between gap-3 flex-wrap text-sm rounded-md p-2" style={{ background: T.bg }}>
                          <div className="flex-1 min-w-[180px]">
                            <div style={{ color: T.text }}>{a.description}</div>
                            <div className="text-xs mt-0.5" style={{ color: T.textMute }}>Owner: {a.owner || "—"} · Due: <span style={{ color: overdue ? T.red : T.textMute, fontWeight: overdue ? 700 : 400 }}>{fmtDate(a.dueDate)}</span>{overdue && " (overdue)"}</div>
                          </div>
                          <div className="flex items-center gap-2">
                            <Select value={a.status} onChange={(e) => setActionStatus(a.id, e.target.value)} style={{ fontSize: 12, padding: "4px 6px" }}>{ACTION_STATUSES.map((s) => <option key={s}>{s}</option>)}</Select>
                            <Pill text={overdue ? "Overdue" : a.status} color={overdue ? { c: T.red, bg: T.redTint } : statusColor(a.status)} />
                            <button onClick={() => setEditingAction(a)} style={{ color: T.ocean }}><Pencil size={12} /></button>
                            <button onClick={() => removeAction(a.id)} style={{ color: T.red, fontSize: 11, fontWeight: 600 }}>×</button>
                          </div>
                        </div>
                      );
                    })}
                  </div>
                )}
                {actionFormFor === f.id && <ActionForm onClose={() => setActionFormFor(null)} onSave={(a) => addAction(f.id, a)} />}
                {editingAction && editingAction.findingId === f.id && <ActionForm onClose={() => setEditingAction(null)} onSave={updateAction} initial={editingAction} />}
              </div>
            );
          })}
        </div>
      )}
      {showFindingForm && <FindingForm onClose={() => setShowFindingForm(false)} onSave={addFinding} themes={settings.themes} onManageThemes={() => { setShowFindingForm(false); setShowThemeManager(true); }} priorFindings={Object.values(allRecords || {}).filter((r) => r.meta.id !== record.meta.id).flatMap((r) => r.findings.map((f) => ({ ...f, engTitle: r.meta.title })))} />}
      {editingFinding && <FindingForm onClose={() => setEditingFinding(null)} onSave={updateFinding} themes={settings.themes} onManageThemes={() => { setEditingFinding(null); setShowThemeManager(true); }} priorFindings={[]} initial={editingFinding} />}
      {showThemeManager && <ThemeManager themes={settings.themes} onSave={onUpdateThemes} onClose={() => setShowThemeManager(false)} />}
    </div>
  );
}
function FindingForm({ onClose, onSave, themes, onManageThemes, priorFindings, initial }) {
  const [f, setF] = useState(initial || { ref: "", title: "", description: "", riskRating: "Medium", theme: themes[0] || "", rootCause: "", findingType: "Recommendation", dateRaised: todayISO() });
  const [err, setErr] = useState(false);
  const set = (k, v) => setF((x) => ({ ...x, [k]: v }));
  const submit = () => { if (!f.title.trim()) { setErr(true); return; } onSave(f); };
  const repeats = (priorFindings || []).filter((p) => p.theme && p.theme === f.theme);
  return (
    <Modal title={initial ? "Edit finding" : "Add finding"} onClose={onClose} wide>
      <div className="flex flex-col gap-3">
        <Field label="Type">
          <Select value={f.findingType || "Recommendation"} onChange={(e) => set("findingType", e.target.value)}>{FINDING_TYPES.map((t) => <option key={t}>{t}</option>)}</Select>
        </Field>

Might be worth flagging as a recurring issue rather than a one-off.</div>

        </Field>

  );

    `This period covered ${filteredIndex.length} engagement(s) with ${allFindings.length} finding(s) raised. Add your key messages for the Group Safety Resilience Board here — headline themes, notable movements, and any areas needing escalation.`

            <div className="text-sm font-semibold" style={{ color: T.textMute }}>Group Compliance & Assurance — Themes & Trends Report</div>

          <div className="text-xs text-right" style={{ color: T.textMute }}>Generated {fmtDate(todayISO())}</div>
        </div>

        </div>

        <div className="grid grid-cols-2 gap-6">
          <ReportSection title="By airport">
            {byAirport.length === 0 ? <ReportEmpty text="No data." /> : byAirport.map(([k, v]) => <BarRow key={k} label={k} value={v} max={Math.max(1, ...byAirport.map((x) => x[1]))} color={T.sky} />)}
          </ReportSection>
          <ReportSection title="By area">
            {byArea.length === 0 ? <ReportEmpty text="No data." /> : byArea.map(([k, v]) => <BarRow key={k} label={k} value={v} max={Math.max(1, ...byArea.map((x) => x[1]))} color={T.navy} />)}

          )}
        </ReportSection>
      </div>
    </div>
  );
}

function ScopeDocumentView({ record, orgName, onUpdate, onBack }) {
  const { meta } = record;
  const emails = record.contacts.map((c) => c.email).filter(Boolean);
  const [aiLoading, setAiLoading] = useState(false);
  const [aiError, setAiError] = useState("");
  const hasContent = !!(meta.background || meta.objective || meta.approach || meta.limitations);

  const draftWithAI = async () => {
    if (hasContent && !window.confirm("This will overwrite the current Background, Objective, Approach and Limitations with an AI-drafted version. Continue?")) return;
    setAiLoading(true);
    setAiError("");
    try {
      const parsed = await runAIScopeDraft(meta.title, meta.airport, meta.area);
      const prefix = meta.controlPrefix || deriveControlPrefix(meta.title);
      onUpdate((r) => {
        const newWorkbook = [];
        const draftIdToRealRef = {};
        if (r.workbook.length === 0) {
          parsed.controls.forEach((c) => {
            const realRef = nextControlRef(newWorkbook, prefix);
            draftIdToRealRef[c.id] = realRef;
            newWorkbook.push({ id: uid("ctl"), ref: realRef, description: c.description, testProcedure: c.test || "", tester: "", result: "In Progress", notes: "AI-drafted from a live web search — verify against the source legislation before relying on this control.", regulatoryRef: c.regulatoryRef || "" });
          });
        }
        const newEvidence = r.workbook.length === 0 && r.evidence.length === 0
          ? parsed.evidence.map((e) => ({ id: uid("ev"), controlRef: (e.controlDraftId && draftIdToRealRef[e.controlDraftId]) || "", description: e.description, requestedFrom: "", dateRequested: todayISO(), dueDate: "", link: "", status: "Not Requested" }))
          : r.evidence;
        return {
          ...r,
          meta: { ...r.meta, background: parsed.background || r.meta.background, objective: parsed.objective || r.meta.objective, approach: parsed.approach || r.meta.approach, limitations: parsed.limitations || r.meta.limitations, controlPrefix: r.meta.controlPrefix || prefix, aiDrafted: true },
          workbook: r.workbook.length === 0 ? newWorkbook : r.workbook,
          evidence: newEvidence,
        };
      });
    } catch (err) {
      console.error("Scope AI-fill failed:", err);
      setAiError(`Couldn't draft the scope just now (${err.message}) — please try again, or fill it in manually on the Overview tab.`);
    } finally {
      setAiLoading(false);
    }
  };

  const plainText = () => {
    const lines = [];
    lines.push(`${meta.title} — Engagement Scope`);
    lines.push(`${meta.airport} · ${meta.area} · ${meta.type}`);
    lines.push("");
    lines.push(`Assurance lead: ${meta.lead || "TBC"}`);
    lines.push(`Timeline: ${fmtDate(meta.startDate)} to ${fmtDate(meta.targetDate)}`);
    lines.push("");
    if (meta.background) { lines.push("BACKGROUND"); lines.push(meta.background); lines.push(""); }
    if (meta.objective) { lines.push("OBJECTIVE & SCOPE"); lines.push(meta.objective); lines.push(""); }
    if (meta.approach) { lines.push("APPROACH"); lines.push(meta.approach); lines.push(""); }
    if (record.workbook.length > 0) {
      lines.push("CONTROLS TO BE TESTED");
      record.workbook.forEach((c) => lines.push(`- [${c.ref}] ${c.description}${c.regulatoryRef ? ` (${c.regulatoryRef})` : ""}`));
      lines.push("");
    }
    if (meta.limitations) { lines.push("EXCEPTIONS & LIMITATIONS"); lines.push(meta.limitations); lines.push(""); }
    return lines.join("\n");
  };

  const emailScope = () => {
    if (emails.length === 0) return;
    const subject = encodeURIComponent(`Engagement scope — ${meta.title}`);
    const body = `Hi,\n\nAhead of fieldwork starting on "${meta.title}" (${meta.airport}, ${meta.area}), please find a summary of the engagement scope below. For the full document with background, approach and the complete control list, use "Print / save as PDF" on the scope page and attach it separately.\n\nObjective: ${meta.objective || "See full scope document"}\n\nTimeline: ${fmtDate(meta.startDate)} to ${fmtDate(meta.targetDate)}\nConducted by: ${meta.lead || "TBC"}\nControls in scope: ${record.workbook.length}\n\nPlease let us know if you have any questions before we begin.\n\nThanks,\n${meta.lead || ""}`;
    openMailOrClipboard(`mailto:${emails.join(",")}?subject=${subject}&body=${encodeURIComponent(body)}`, body, "Scope emailed to stakeholders");
  };

  return (
    <div className="print-page" style={{ background: T.bg, minHeight: "100vh" }}>
      <div className="no-print flex items-center justify-between px-6 py-4 max-w-[820px] mx-auto flex-wrap gap-2">
        <button onClick={onBack} className="flex items-center gap-1.5 text-sm font-semibold" style={{ color: T.ocean }}><ArrowLeft size={15} /> Back to engagement</button>
        <div className="flex gap-2">
          <Button variant="ghost" onClick={draftWithAI} style={{ opacity: aiLoading ? 0.6 : 1 }}>{aiLoading ? "Drafting…" : <><Sparkles size={15} /> {hasContent ? "Re-draft with AI" : "Draft with AI"}</>}</Button>
          <Button variant="ghost" onClick={emailScope} style={{ opacity: emails.length === 0 ? 0.5 : 1 }}><Send size={15} /> Email to stakeholders</Button>
          <Button onClick={() => window.print()}><Printer size={15} /> Print / save as PDF</Button>
        </div>
      </div>
      {aiError && <p className="no-print text-xs text-center mb-3 font-semibold" style={{ color: T.red }}>{aiError}</p>}
      {emails.length === 0 && <p className="no-print text-xs text-center mb-3" style={{ color: T.textMute }}>Add key contacts with an email address to enable "Email to stakeholders".</p>}

      <div className="mx-auto" style={{ maxWidth: 820, background: "white", padding: "40px 48px", boxShadow: "0 1px 4px rgba(4,30,66,0.08)" }}>
        <div className="flex items-center justify-between pb-5 mb-6" style={{ borderBottom: `3px solid transparent`, borderImage: `linear-gradient(90deg, ${T.navy}, ${T.teal}) 1` }}>
          <div>
            <div className="text-xs font-bold tracking-widest" style={{ color: T.ocean, letterSpacing: 2 }}>{(orgName || "").toUpperCase()}</div>
            <div className="text-sm font-semibold" style={{ color: T.textMute }}>Group Compliance & Assurance — Engagement Scope</div>
          </div>
          <div className="text-xs text-right" style={{ color: T.textMute }}>Generated {fmtDate(todayISO())}</div>
        </div>

        <h1 className="text-2xl font-extrabold mb-2" style={{ color: T.navy }}>{meta.title}</h1>
        <div className="flex items-center gap-2 mb-6 flex-wrap">
          <Pill text={meta.airport} color={{ c: T.ocean, bg: T.skyTint }} />

          <Pill text={meta.type} color={{ c: T.textMute, bg: T.grayTint }} />

        <div className="grid grid-cols-3 gap-4 mb-7 text-sm">
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Conducted by</div><div style={{ color: T.text }}>{meta.lead || "TBC"}</div></div>
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Fieldwork start</div><div style={{ color: T.text }}>{fmtDate(meta.startDate)}</div></div>
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Target completion</div><div style={{ color: T.text }}>{fmtDate(meta.targetDate)}</div></div>
        </div>

        <ReportSection title="Background"><p className="text-sm" style={{ color: T.text }}>{meta.background || "Not yet recorded — add this on the Overview tab."}</p></ReportSection>
        <ReportSection title="Objective & scope"><p className="text-sm" style={{ color: T.text }}>{meta.objective || "Not yet recorded."}</p></ReportSection>
        <ReportSection title="Approach"><p className="text-sm" style={{ color: T.text }}>{meta.approach || "Not yet recorded — add this on the Overview tab."}</p></ReportSection>

        <ReportSection title="Controls to be tested">
          {record.workbook.length === 0 ? <ReportEmpty text="No controls added to scope yet." /> : (
            <ReportTable cols={["Ref", "Control", "Regulatory reference"]} rows={record.workbook.map((c) => [c.ref, c.description, c.regulatoryRef || "—"])} />
          )}
        </ReportSection>

        <ReportSection title="Exceptions & limitations" last>
          <p className="text-sm" style={{ color: T.text }}>{meta.limitations || "None recorded."}</p>
        </ReportSection>
      </div>
    </div>
  );
}

function ReportView({ record, orgName, onBack }) {

  const openActions = record.actions.filter((a) => a.status !== "Closed");
  return (
    <div className="print-page" style={{ background: T.bg, minHeight: "100vh" }}>
      <div className="no-print flex items-center justify-between px-6 py-4 max-w-[820px] mx-auto">
        <button onClick={onBack} className="flex items-center gap-1.5 text-sm font-semibold" style={{ color: T.ocean }}><ArrowLeft size={15} /> Back to engagement</button>
        <Button onClick={() => window.print()}><Printer size={15} /> Print / save as PDF</Button>
      </div>

      <div className="mx-auto" style={{ maxWidth: 820, background: "white", padding: "40px 48px", boxShadow: "0 1px 4px rgba(4,30,66,0.08)" }}>
        <div className="flex items-center justify-between pb-5 mb-6" style={{ borderBottom: `3px solid transparent`, borderImage: `linear-gradient(90deg, ${T.navy}, ${T.teal}) 1` }}>
          <div>
            <div className="text-xs font-bold tracking-widest" style={{ color: T.ocean, letterSpacing: 2 }}>{(orgName || "").toUpperCase()}</div>
            <div className="text-sm font-semibold" style={{ color: T.textMute }}>Group Compliance & Assurance — Review Summary</div>
          </div>
          <div className="text-xs text-right" style={{ color: T.textMute }}>Generated {fmtDate(todayISO())}</div>
        </div>

        {meta.approvalStatus !== "Approved" && (
          <div className="rounded-lg p-3 mb-5 text-xs font-semibold" style={{ background: T.amberTint, color: T.amber }}>
            This engagement has not yet been reviewed and approved for issue. Treat this as a working draft, not a final Review Summary.
          </div>
        )}

        <h1 className="text-2xl font-extrabold mb-2" style={{ color: T.navy }}>{meta.title}</h1>
        <div className="flex items-center gap-2 mb-6 flex-wrap">
          <Pill text={meta.airport} color={{ c: T.ocean, bg: T.skyTint }} />
          {meta.area && <Pill text={meta.area} color={{ c: T.navy, bg: T.grayTint }} />}
          <Pill text={meta.type} color={{ c: T.textMute, bg: T.grayTint }} />
          <Pill text={meta.stage} color={{ c: T.green, bg: T.greenTint }} />
        </div>

        <div className="grid grid-cols-3 gap-4 mb-6 text-sm">
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Assurance lead</div><div style={{ color: T.text }}>{meta.lead || "—"}</div></div>
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Start date</div><div style={{ color: T.text }}>{fmtDate(meta.startDate)}</div></div>
          <div><div className="text-xs font-semibold" style={{ color: T.textMute }}>Target completion</div><div style={{ color: T.text }}>{fmtDate(meta.targetDate)}</div></div>
        </div>

        <ReportSection title="Background">
          <p className="text-sm" style={{ color: T.text }}>{meta.background || `This ${meta.type.toLowerCase()} was conducted by Group Compliance & Assurance as part of the ${meta.area} second-line assurance programme at ${airportName(meta.airport)}. (Add background detail on the Overview tab before issuing.)`}</p>
        </ReportSection>

        <ReportSection title="Objective & scope"><p className="text-sm" style={{ color: T.text }}>{meta.objective || "No objective recorded."}</p></ReportSection>

        <ReportSection title="Approach">
          <p className="text-sm" style={{ color: T.text }}>{meta.approach || `The review followed MAG's standard second-line assurance process: a self-assessment issued to stakeholders ahead of fieldwork, a joint on-site review of evidence against ${record.workbook.length} control(s), and independent quality review prior to sign-off. (Add approach detail on the Overview tab before issuing.)`}</p>
        </ReportSection>

        <ReportSection title="Conclusion">
          <p className="text-sm" style={{ color: T.text }}>
            {record.workbook.length === 0 ? "No controls were tested." : (() => {
              const counts = CONTROL_RESULTS.filter((r) => r !== "Not Started" && r !== "In Progress").map((r) => [r, record.workbook.filter((c) => c.result === r).length]).filter(([, n]) => n > 0);
              return `Of the ${record.workbook.length} control(s) tested: ${counts.map(([r, n]) => `${n} rated ${r}`).join("; ") || "none yet rated"}. ${record.findings.length} finding(s) were raised, with ${record.actions.filter((a) => a.status !== "Closed").length} action(s) currently open.`;
            })()}
          </p>
        </ReportSection>

        <ReportSection title="Workbook summary">
          {record.workbook.length === 0 ? <ReportEmpty text="No controls recorded." /> : (
            <ReportTable cols={["Ref", "Control", "Result"]} rows={record.workbook.map((c) => [c.ref, c.description, <Pill text={c.result} color={statusColor(c.result)} />])} />
          )}
        </ReportSection>

        <ReportSection title="Evidence reviewed">
          {record.evidence.length === 0 ? <ReportEmpty text="No evidence requests recorded." /> : (
            <ReportTable cols={["Control ref", "Evidence", "Status"]} rows={record.evidence.map((e) => [e.controlRef || "—", e.description, <Pill text={e.status} color={statusColor(e.status)} />])} />
          )}
        </ReportSection>

        <ReportSection title="Detailed findings & recommendations">
          {record.findings.length === 0 ? <ReportEmpty text="No findings raised." /> : (
            <div className="flex flex-col gap-3">
              {record.findings.map((f) => (
                <div key={f.id} style={{ borderLeft: `3px solid ${riskColor(f.riskRating).c}`, paddingLeft: 12 }}>
                  <div className="flex items-center gap-2 mb-0.5 flex-wrap">
                    <span className="text-xs font-bold" style={{ fontFamily: "JetBrains Mono, monospace", color: T.navy }}>{f.ref}</span>
                    <Pill text={f.riskRating} color={riskColor(f.riskRating)} />
                    {f.theme && <Pill text={f.theme} color={{ c: T.ocean, bg: T.skyTint }} />}
                  </div>
                  <div className="text-sm font-bold" style={{ color: T.text }}>{f.title}</div>
                  <div className="text-xs mt-0.5" style={{ color: T.textMute }}>{f.description}</div>
                  {f.rootCause && <div className="text-xs mt-1" style={{ color: T.text }}><strong>Root cause:</strong> {f.rootCause}</div>}
                  {record.actions.filter((a) => a.findingId === f.id).length > 0 && (
                    <div className="text-xs mt-1" style={{ color: T.textMute }}>
                      <strong>Agreed action(s):</strong> {record.actions.filter((a) => a.findingId === f.id).map((a) => `${a.description} (${a.owner || "unassigned"}, due ${fmtDate(a.dueDate)})`).join("; ")}
                    </div>
                  )}
                </div>
              ))}
            </div>
          )}
        </ReportSection>

        <ReportSection title="Actions">
          {record.actions.length === 0 ? <ReportEmpty text="No actions raised." /> : (
            <ReportTable cols={["Action", "Owner", "Due", "Status"]} rows={record.actions.map((a) => [a.description, a.owner || "—", fmtDate(a.dueDate), <Pill text={isOverdue(a.dueDate, a.status) ? "Overdue" : a.status} color={isOverdue(a.dueDate, a.status) ? { c: T.red, bg: T.redTint } : statusColor(a.status)} />])} />
          )}
          {openActions.length > 0 && <p className="text-xs mt-3" style={{ color: T.textMute }}>{openActions.length} action(s) remain open at time of report generation.</p>}
        </ReportSection>

        <div style={{ marginTop: 30, paddingTop: 20, borderTop: `1px solid ${T.border}` }}>
          <h3 className="text-sm font-bold uppercase tracking-wide mb-3" style={{ color: T.navy, letterSpacing: 0.5 }}>Reviewed & approved for issue by (must not be the drafter)</h3>
          <div className="grid grid-cols-2 gap-6 text-sm" style={{ color: T.text }}>
            <div>Name: {meta.approver || "________________________"}</div>
            <div>Date: {meta.approvalDate ? fmtDate(meta.approvalDate) : "________________________"}</div>
          </div>
        </div>
      </div>
    </div>
  );
}
function ReportSection({ title, children, last }) {
  return <div style={{ marginBottom: last ? 0 : 26 }}><h3 className="text-sm font-bold uppercase tracking-wide mb-2.5" style={{ color: T.navy, letterSpacing: 0.5 }}>{title}</h3>{children}</div>;
}
function ReportEmpty({ text }) { return <p className="text-xs italic" style={{ color: T.textMute }}>{text}</p>; }
function ReportTable({ cols, rows }) {
  return (
    <table className="w-full text-xs" style={{ borderCollapse: "collapse" }}>
      <thead><tr>{cols.map((c, i) => <th key={i} className="text-left py-1.5 font-semibold" style={{ color: T.navy, borderBottom: `1px solid ${T.border}` }}>{c}</th>)}</tr></thead>
      <tbody>{rows.map((r, ri) => <tr key={ri}>{r.map((cell, ci) => <td key={ci} className="py-1.5 pr-3 align-top" style={{ color: T.text, borderBottom: `1px solid ${T.grayTint}` }}>{cell}</td>)}</tr>)}</tbody>
    </table>
  );
}
